1# SOLAR-ARCH-013 — toda entrada que transporta dados externos passa 

2# por Validator e Sanitizer. 

3# 

4# A regra lê o `HttpInput` que o `http_input_scanner` produziu a partir 

5# das rotas. O critério é TRANSPORTE DE DADOS (`transports_data()`), e 

6# não o verbo HTTP: é isso que faz uma rota nova (verbo novo, tipo de 

7# conteúdo novo) já nascer coberta, sem alterar a regra. 

8from nucleus.exceptions.solar_exception import SolarException 

9from nucleus.rules.rule import Rule 

10from nucleus.contracts.http_input import HttpInput 

11 

12 

13class HttpInputMustPassThroughSanitizationAndValidation(Rule): 

14 """ 

15 Regra arquitetural: 

16 

17 SOLAR-ARCH-013 

18 

19 Toda entrada HTTP que TRANSPORTA DADOS EXTERNOS deve passar 

20 pelo pipeline de Sanitization e Validation apropriado ao tipo 

21 de conteúdo recebido. 

22 

23 Fluxo exigido: 

24 

25 HTTP Input → Validator → Sanitizer → Planet 

26 

27 A ordem importa: a validação confere se os campos esperados 

28 EXISTEM, e só depois o Sanitizer limpa os valores. Sanitizar 

29 primeiro seria limpar campo que pode não ter chegado. 

30 

31 O critério é o TRANSPORTE DE DADOS EXTERNOS, não o verbo HTTP. 

32 Portanto a regra NÃO é "POST exige validação": 

33 

34 POST /documents multipart/form-data → exige pipeline 

35 POST /users application/json → exige pipeline 

36 PUT /users application/json → exige pipeline 

37 PATCH /users application/json → exige pipeline 

38 POST /users form-urlencoded → exige pipeline 

39 GET /solar/health (sem corpo) → NÃO exige 

40 

41 O Sanitizer e o Validator são específicos ao tipo de conteúdo 

42 (JsonSanitizer/JsonValidator, FileSanitizer/FileValidator, 

43 FormSanitizer/FormValidator, ...). A regra exige que EXISTAM; 

44 qual implementação é apropriada é decisão do contrato. 

45 

46 ESCOPO — o que esta regra NÃO cobre: 

47 

48 A regra NÃO valida regra de negócio. Validações estruturais 

49 de entrada (arquivo enviado, tamanho, extensão, MIME type, 

50 quantidade, arquivo vazio) pertencem ao Validator. 

51 

52 Regras de negócio ("o documento precisa representar um 

53 contrato válido com determinado número de contrato") ficam no 

54 UseCase/Domain e NÃO devem ser transferidas ao Validator para 

55 satisfazer esta regra. 

56 

57 Exemplo válido: 

58 

59 HttpInput( 

60 method="POST", 

61 name="/documents", 

62 sanitizer=FileSanitizer(), 

63 validator=FileValidator(), 

64 ) 

65 

66 O contexto desta regra é um HttpInput — diferente do contexto 

67 das regras de Dependency (SOLAR-ARCH-001 a 012). 

68 """ 

69 

70 def code(self) -> str: 

71 return "SOLAR-ARCH-013" 

72 

73 def description(self) -> str: 

74 return ( 

75 "Toda entrada HTTP que transporta dados externos deve " 

76 "passar por Sanitization e Validation apropriadas ao " 

77 "tipo de conteúdo recebido." 

78 ) 

79 

80 def validate(self, http_input: HttpInput) -> None: 

81 

82 # A regra só se aplica a entradas que transportam 

83 # dados externos. 

84 if not http_input.transports_data(): 

85 return 

86 

87 # Recolhe TUDO o que falta antes de acusar: a mensagem cita os 

88 # dois guardas de uma vez, e quem corrige não precisa rodar o 

89 # check duas vezes para descobrir o segundo. 

90 missing = self._missing_guards(http_input) 

91 

92 if not missing: 

93 return 

94 

95 raise SolarException( 

96 self._build_message(http_input, missing) 

97 ) 

98 

99 def _missing_guards( 

100 self, 

101 http_input: HttpInput 

102 ) -> list[str]: 

103 

104 missing: list[str] = [] 

105 

106 # Checagem por AUSÊNCIA e não por presença: `str` (o nome 

107 # encontrado na análise estática) também satisfaz o contrato 

108 # — ver `HttpInput.sanitizer`. Só `None` acusa. 

109 if http_input.sanitizer is None: 

110 missing.append("Sanitizer") 

111 

112 if http_input.validator is None: 

113 missing.append("Validator") 

114 

115 return missing 

116 

117 def _build_message( 

118 self, 

119 http_input: HttpInput, 

120 missing: list[str] 

121 ) -> str: 

122 

123 return ( 

124 f"{self.code()}: " 

125 f"a entrada HTTP '{http_input.method} " 

126 f"{http_input.name}' não possui " 

127 f"{' e '.join(missing)}. " 

128 f"Toda entrada HTTP que transporta dados externos deve " 

129 f"passar por Sanitizer e Validator apropriados ao tipo " 

130 f"de conteúdo." 

131 )

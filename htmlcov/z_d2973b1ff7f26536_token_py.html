1""" 

2Emissão e verificação de token JWT. 

3 

4O token substitui a sessão guardada no servidor: ele carrega a 

5identidade (`sub`) e a validade (`exp`), assinado com a SECRET_KEY 

6da aplicação. Se alguém alterar o conteúdo, a assinatura deixa de 

7bater e a verificação recusa (SOLAR-ARCH-027). 

8""" 

9 

10import jwt 

11 

12from datetime import datetime, timedelta, timezone 

13 

14 

15class Token: 

16 

17 def __init__(self, secret_key: str, expires_hours: int): 

18 self.secret_key = secret_key 

19 self.expires_hours = expires_hours 

20 

21 def create(self, id_user: int) -> str: 

22 """ 

23 Emite um token para o usuário. 

24 """ 

25 

26 expire_time = timedelta(hours=self.expires_hours) 

27 

28 payload = { 

29 # `sub` é padronizado, e vai como STRING: o JWT não 

30 # aceita número nesta chave. 

31 "sub": str(id_user), 

32 

33 # `exp` é o prazo. Um token vazado deixa de servir 

34 # sozinho quando o prazo passa — é o que limita a janela 

35 # de uso de um token roubado. 

36 "exp": datetime.now(timezone.utc) + expire_time 

37 } 

38 

39 # Algoritmo EXPLÍCITO (SOLAR-ARCH-027). Um `encode` sem 

40 # algoritmo declarado pode cair em token sem assinatura 

41 # nenhuma. 

42 return jwt.encode( 

43 payload, 

44 self.secret_key, 

45 algorithm="HS256" 

46 ) 

47 

48 def verify(self, token: str) -> dict: 

49 """ 

50 Confere o token e devolve o conteúdo. 

51 """ 

52 

53 if not token: 

54 raise ValueError("Token não fornecido") 

55 

56 # Aceita o formato do cabeçalho HTTP ("Bearer <token>") e o 

57 # token puro — quem chama não precisa decidir qual dos dois 

58 # chegou. 

59 if token.startswith("Bearer "): 

60 token = token[7:] 

61 

62 try: 

63 # `algorithms` EXPLÍCITO na verificação. 

64 # 

65 # Sem a lista, a biblioteca aceitaria o algoritmo 

66 # declarado NO PRÓPRIO token — e bastaria dizer 

67 # `alg: none` para entrar sem assinatura 

68 # (SOLAR-ARCH-027). 

69 return jwt.decode( 

70 token, 

71 self.secret_key, 

72 algorithms=["HS256"] 

73 ) 

74 

75 # Expirado e inválido viram mensagens DIFERENTES de 

76 # propósito: quem está com a tela aberta precisa saber que é 

77 # só entrar de novo, e não que a sessão foi corrompida. 

78 except jwt.ExpiredSignatureError: 

79 raise ValueError( 

80 "Sessão expirada. Faça login novamente" 

81 ) 

82 

83 except jwt.InvalidTokenError: 

84 raise ValueError( 

85 "Token inválido" 

86 )

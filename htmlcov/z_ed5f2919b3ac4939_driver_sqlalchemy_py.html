51class DriverSQLAlchemy(DatabaseDriver): 

52 """ 

53 Driver de persistência da SOLAR baseado em SQLAlchemy. 

54 

55 É responsabilidade deste driver: 

56 

57 - interpretar DB_CONNECTION; 

58 - resolver o dialect/driver do SQLAlchemy; 

59 - montar a URL de conexão. 

60 

61 O Environment apenas fornece os valores brutos do ambiente. 

62 """ 

63 

64 # O que REMOVE alguma coisa do banco. 

65 # 

66 # As quatro sao listadas uma a uma porque no Alembic elas NAO 

67 # tem uma base comum: cada uma herda por conta propria, e um 

68 # `isinstance` num ancestral unico nao existe. Uma remocao fora 

69 # desta lista ficaria no banco para sempre, mesmo com `prune`. 

70 REMOVAL_OPERATIONS = ( 

71 DropTableOp, 

72 DropColumnOp, 

73 DropIndexOp, 

74 DropConstraintOp, 

75 DropTableCommentOp, 

76 ) 

77 

78 # Traduz o que está no ambiente para o que a ORM entende. 

79 # 

80 # A aplicação escreve `DB_CONNECTION=postgres`; quem sabe que 

81 # isso vira `postgresql+psycopg` é o driver — é por isso que o 

82 # Environment devolve o valor CRU, sem interpretar. 

83 __CONNECTION_CONFIG = { 

84 "postgres": { 

85 "dialect": "postgresql", 

86 "driver": "psycopg", 

87 }, 

88 "sqlite": { 

89 "dialect": "sqlite", 

90 "driver": "pysqlite", 

91 },

118 def _build_connection_url(self): 

119 """ 

120 Monta a URL de conexão a partir da configuração do ambiente. 

121 

122 O dialect/driver é resolvido a partir de DB_CONNECTION e a 

123 URL é montada internamente com os demais parâmetros 

124 (DB_NAME, DB_HOST, DB_PORT, DB_USER, DB_PASS). 

125 """ 

126 

127 connection = self.config.get("DB_CONNECTION") 

128 

129 database = self.__CONNECTION_CONFIG.get(connection) 

130 

131 if database is None: 

132 raise DatabaseException( 

133 f"Conexão de banco '{connection}' não suportada.", 

134 code="INVALID_DATABASE_CONNECTION" 

135 ) 

136 

137 dialect = database["dialect"] 

138 driver = database["driver"] 

139 

140 # SQLite não tem host, porta, usuário nem senha: a URL é o 

141 # caminho do arquivo — ou `:memory:`, que é o que o 

142 # ambiente TEST usa para rodar sem banco nenhum. 

143 if dialect == "sqlite": 

144 return ( 

145 f"{dialect}" 

146 f"+{driver}:///" 

147 f"{self.config['DB_NAME']}" 

148 ) 

149 

150 # `quote_plus` porque usuário e senha viram parte da URL: 

151 # uma senha com `@`, `:` ou `/` montaria uma URL quebrada 

152 # — ou, pior, apontando para outro host. 

153 user = quote_plus(str(self.config["DB_USER"])) 

154 password = quote_plus(str(self.config["DB_PASS"])) 

155 

156 return ( 

157 f"{dialect}" 

158 f"+{driver}://" 

159 f"{user}:{password}@" 

160 f"{self.config['DB_HOST']}:" 

161 f"{self.config['DB_PORT']}/" 

162 f"{self.config['DB_NAME']}" 

163 )

166 # Sessão primeiro, engine depois: a sessão usa o engine. 

167 if self._session:

169 self._session = None

173 if self.engine: 

174 self.engine.dispose() 

175 self.engine = None

218 def sync(self, *, prune=True): 

219 """ 

220 Sincroniza o esquema do banco com os Models. 

221 

222 Duas etapas, nesta ordem: 

223 

224 1. cria o que ainda NAO existe — a tabela inteira, com 

225 indices e chaves, como o Model declara; 

226 2. altera o que JA EXISTE e mudou: coluna nova, tipo 

227 diferente, obrigatoriedade diferente. 

228 

229 A ordem nao e gosto: a comparacao da etapa 2 e feita 

230 DEPOIS da etapa 1, entao o que a etapa 1 criou nao 

231 aparece mais como diferenca. 

232 

233 `prune` REMOVE o que existe no banco e nao existe mais 

234 no Model — coluna, indice ou tabela. E o padrao: um 

235 esquema que so CRESCE acumula lixo, e a coluna que saiu 

236 do Model continua aceitando escrita de quem nao sabe que 

237 ela deixou de existir. 

238 """ 

239 

240 if self.engine is None: 

241 raise DatabaseException( 

242 "O driver SQLAlchemy não está conectado.", 

243 code="DATABASE_NOT_CONNECTED" 

244 ) 

245 

246 # So as tabelas dos Models que foram IMPORTADOS: a base da 

247 # ORM conhece o que ja passou pelo import. Quem importa e a 

248 # fachada `Database`, antes de chegar aqui. 

249 Base.metadata.create_all(self.engine) 

250 

251 self._apply_changes(prune=prune)

253 def _apply_changes(self, *, prune=False): 

254 """ 

255 Aplica no banco o que os Models pedem e ele ainda nao tem. 

256 """ 

257 

258 # `begin()`: o que a comparacao encontrar e aplicado numa 

259 # transacao so. Se uma alteracao falhar, nenhuma fica pela 

260 # metade — e o esquema nao vira um meio-termo que ninguem 

261 # consegue explicar depois. 

262 with self.engine.begin() as connection: 

263 

264 # `compare_type`: sem isso o diff so olharia nomes de 

265 # coluna, e uma coluna que mudou de tipo passaria batido. 

266 context = MigrationContext.configure( 

267 connection, 

268 opts={"compare_type": True} 

269 ) 

270 

271 # Quem compara e o Alembic. Reimplementar este diff a mao 

272 # seria a receita para um dia apagar dado por engano. 

273 script = produce_migrations(context, Base.metadata) 

274 

275 operations = [ 

276 operation 

277 for operation in self._flatten(script.upgrade_ops.ops) 

278 if prune or not isinstance(operation, self.REMOVAL_OPERATIONS) 

279 ] 

280 

281 # Nada mudou: nao ha o que executar, e a subida segue. 

282 if not operations: 

283 return 

284 

285 runner = Operations(context) 

286 

287 for operation in operations: 

288 runner.invoke(operation) 

289 

290 @staticmethod 

291 def _flatten(operations): 

292 """ 

293 Abre os grupos por tabela numa lista so. 

294 

295 O Alembic agrupa as alteracoes de uma mesma tabela em 

296 `ModifyTableOps`, que e um AGRUPAMENTO e nao uma 

297 operacao: ele nao pode ser executado, so o que esta 

298 dentro dele. 

299 """ 

300 

301 for operation in operations: 

302 

303 if isinstance(operation, ModifyTableOps): 

304 yield from DriverSQLAlchemy._flatten(operation.ops) 

305 continue 

306 

307 yield operation

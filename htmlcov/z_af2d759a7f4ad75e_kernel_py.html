48class Kernel: 

49 """ 

50 Monta a aplicação inteira: configuração, mecanismos, rotas e 

51 observação. 

52 

53 A ordem da montagem é a ordem das dependências: 

54 

55 Environment de onde vêm os valores 

56 Identity quem a aplicação é (SECRET_KEY, SOLAR_ID) 

57 Drivers a tecnologia (banco, arquivos, eventos) 

58 Middleware as proteções de entrada e de saída 

59 Rotas o ponto central orbits/routes.py, que agrega 

60 as dos Planets 

61 

62 `mount()` devolve a aplicação pronta. Subir o servidor NÃO é 

63 papel do Kernel: quem sobe é o Server, que recebe a aplicação. 

64 

65 O Kernel também NÃO conhece o Observatório — conhece o gancho 

66 `observe()`. A direção da dependência é Observatory → Nucleus, 

67 então quem liga os dois é o ponto de montagem, fora dos dois. 

68 """ 

69 

70 def __init__( 

71 self, 

72 env_file=None, 

73 config=None, 

74 storage_root=None 

75 ): 

76 """ 

77 env_file: 

78 Arquivo de ambiente (SECRET_KEY, SOLAR_ID). Por padrão, 

79 `nucleus/.env`. 

80 

81 config: 

82 Configuração já lida, para quando ela vem de fora (nos 

83 testes, por exemplo). Sem ela, quem lê é o Environment. 

84 

85 storage_root: 

86 Raiz dos arquivos da aplicação. Por padrão, `storage/` 

87 na raiz do projeto. 

88 """ 

89

91 Path(env_file) if env_file else DEFAULT_ENV_FILE 

92 )

106 self.identity = None

112 self.response_pipeline = None 

113 self.error_handler = None 

114 

115 # A aplicação montada, quando houver uma. 

116 self.app = None

122 def mount(self) -> Flask: 

123 """ 

124 Monta a aplicação e devolve pronta para subir. 

125 

126 O corpo desta função É a receita da montagem, na ordem. Ler 

127 este bloco de cima para baixo é ler o que acontece quando a 

128 aplicação sobe. 

129 

130 Subir o servidor NÃO é papel daqui — quem sobe é o Server. 

131 """ 

132 

133 # AMBIENTE — de onde vêm os valores 

134 # 

135 # O Environment lê o arquivo .env filtrando pelo ambiente 

136 # ativo: com APP_ENV=DEV ele lê DEV_DB_HOST, DEV_PORT, e 

137 # assim por diante. O que volta são valores CRUS — quem 

138 # interpreta cada um é o driver, não aqui. 

139 self.config = self._config or Environment.init() 

140 

141 # IDENTIDADE — quem a aplicação é 

142 # 

143 # Garante SECRET_KEY (segredo que assina sessão e token) e 

144 # SOLAR_ID (identificador NÃO secreto, usado em log e 

145 # auditoria). Se ainda não existirem no arquivo de ambiente, 

146 # são gerados e GRAVADOS nele — é o efeito colateral que faz 

147 # a montagem não poder rodar duas vezes ao mesmo tempo. 

148 self.identity = Identity(env_file=str(self.env_file)) 

149 self.identity.init() 

150 

151 # BANCO DE DADOS 

152 # 

153 # A aplicação fala com a fachada `Database` e NUNCA com o 

154 # driver (SOLAR-ARCH-008). O que o driver monta — dialect, 

155 # URL de conexão, engine — é decisão dele, a partir do 

156 # config cru que veio do ambiente. 

157 # 

158 # `init` CONECTA e MONTA AS ESTRUTURAS (§8.1): o banco cuida 

159 # do próprio esquema. Este arquivo não sabe de Model, de 

160 # tabela nem de sincronização — e quem quiser desligar a 

161 # montagem desliga no `Database` (`SYNC_ON_START`). 

162 Database.init(DriverSQLAlchemy(self.config)) 

163 

164 # PERSISTÊNCIA — instâncias do Planet, DEPOIS do banco 

165 # 

166 # O Kernel inicializa a fachada `Database` e é quem a encerra 

167 # no shutdown. As instâncias prontas moram nos próprios 

168 # módulos do Planet (Persistence → Repository → UseCase), 

169 # importadas junto com as rotas logo abaixo. Esse import só 

170 # pode acontecer DEPOIS de `Database.init()`, porque os 

171 # módulos de Persistence importam o Model no topo 

172 # (`class Users(...)` exige a ORM publicada). 

173 # 

174 # Não existe registry genérico de persistências: cada Planet 

175 # conhece apenas a Persistence do Model de que precisa. 

176 

177 # ARMAZENAMENTO DE ARQUIVOS 

178 # 

179 # Mesma ideia das outras fachadas: a aplicação chama 

180 # `Storage.save/read` e não sabe onde o arquivo foi parar. 

181 # Aqui, gravação no disco da própria aplicação. 

182 Storage.init(LocalStorageDriver(self.storage_root)) 

183 

184 # EVENTOS 

185 # 

186 # Fachada de publish/subscribe. O driver em memória atende 

187 # UM processo: trocar por um broker é trocar esta linha, sem 

188 # tocar em quem publica. 

189 Broadcast.init(InMemoryBroadcast()) 

190 

191 # MIDDLEWARE — as proteções de entrada, saída e erro 

192 # 

193 # Ficam aqui, e não espalhadas por rota: proteção que 

194 # depende de alguém lembrar de declarar é proteção que uma 

195 # hora falha. As rotas USAM estes mecanismos; não os criam. 

196 self.security = SecurityPolicy() 

197 self.request_pipeline = RequestPipeline() 

198 self.error_handler = ErrorHandler() 

199 

200 # APLICAÇÃO 

201 app = Flask("solar") 

202 

203 # A mesma chave da identidade assina a sessão do Flask. É o 

204 # que impede cookie de sessão forjado. 

205 app.secret_key = self.identity.secret_key 

206 

207 # CABEÇALHOS DE SEGURANÇA — em toda resposta 

208 # 

209 # Os obrigatórios (nosniff, DENY, no-referrer) vão em 

210 # QUALQUER resposta, inclusive nas de erro: um 500 com 

211 # conteúdo e sem nosniff também é brecha. 

212 headers = SecurityHeaders(self.security) 

213 

214 @app.after_request 

215 def apply_security_headers(response): 

216 

217 for name, value in headers.headers().items(): 

218 

219 # Sobrepõe o que a rota tiver definido: ninguém 

220 # desliga a proteção respondendo diferente. 

221 response.headers[name] = value 

222 

223 return response 

224 

225 # ERROS — mensagem genérica para quem chamou, e o detalhe 

226 # inteiro para os observadores 

227 # 

228 # Registrado em Exception, e não só nas exceções da SOLAR: 

229 # falha inesperada também vira resposta segura, em vez de 

230 # página de erro do servidor. 

231 @app.errorhandler(Exception) 

232 def handle_error(error): 

233 

234 # A tradução de exceção em código HTTP fica aqui, e não 

235 # na exceção: a exceção fala da regra quebrada, o código 

236 # fala com o cliente. 

237 # 

238 # 400 erro de quem chamou (entrada inválida, contrato) 

239 # 403 segurança (token, URL, permissão) 

240 # 500 o resto — e o resto NUNCA diz o que foi 

241 status = 500 

242 

243 if isinstance(error, (ValidationException, ContractException)): 

244 status = 400 

245 

246 elif isinstance(error, SecurityException): 

247 status = 403 

248 

249 # O payload que sai NÃO leva o detalhe interno: leva 

250 # código estável, mensagem genérica e a referência que 

251 # liga esta resposta ao registro (SOLAR-ARCH-028). 

252 # 

253 # O contexto vai junto para o registro: método, rota e 

254 # status são o que permite investigar depois. 

255 payload = self.error_handler.handle( 

256 error, 

257 { 

258 "method": request.method, 

259 "route": request.path, 

260 "status": status, 

261 }, 

262 ) 

263 

264 return jsonify(payload), status 

265 

266 # ROTAS — o ponto central agrega as dos Planets 

267 # 

268 # O Kernel não conhece Planet nenhum: ele registra o que o 

269 # ponto central montou. Planet novo não muda este arquivo. 

270 # 

271 # O import vem DEPOIS de `Database.init()` pelo mesmo motivo 

272 # da Persistence: a cadeia Controller → UseCase → Repository → 

273 # Persistence → Model também depende da ORM publicada. 

274 from orbits.routes import routes as orbit_routes 

275 

276 app.register_blueprint(orbit_routes()) 

277 

278 # MECANISMOS — deixados à mão de quem está dentro de uma 

279 # rota, para não recriar o que já está montado. 

280 app.extensions["solar"] = self 

281 

282 self.app = app 

283 

284 return app

286 def observe(self, *observers) -> None: 

287 """ 

288 Liga observadores ao tratamento de erro. 

289 

290 Cada observador é um callable (erro, payload, contexto) — 

291 o gancho por onde o Observatório recebe os erros sem que o 

292 Nucleus precise conhecê-lo. 

293 """ 

294 

295 self.error_handler.observe(*observers)

297 def shutdown(self) -> None: 

298 """ 

299 Encerra o que segura recurso. 

300 """ 

301

303 # banco segura, e é ele que precisa ser fechado. As 

304 # Persistences dos Planets guardam só a fachada `Database`, 

305 # nunca a sessão, então não há estado obsoleto a descartar. 

306 Database.close()

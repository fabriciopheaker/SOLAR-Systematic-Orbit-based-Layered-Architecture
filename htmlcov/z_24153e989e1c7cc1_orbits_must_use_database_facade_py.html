1# SOLAR-ARCH-008 — Orbits fala com a fachada, não com o Driver. 

2# 

3# A fachada `nucleus.database.database` é o único ponto por onde a 

4# infraestrutura de persistência pode ser alcançada de fora do 

5# Nucleus. É ela que permite trocar o driver sem tocar nos Orbits. 

6from nucleus.rules.scanner.dependency import Dependency 

7from nucleus.exceptions.solar_exception import SolarException 

8from nucleus.rules.rule import Rule 

9 

10 

11class OrbitsMustUseDatabaseFacade(Rule): 

12 """ 

13 Regra arquitetural: 

14 

15 SOLAR-ARCH-008 

16 

17 O Database do Nucleus é a fachada da infraestrutura 

18 de persistência. 

19 

20 Direção permitida: 

21 

22 Orbits → Nucleus.Database → Driver 

23 

24 Os Orbits NÃO devem conhecer o Driver diretamente. 

25 Todo acesso à infraestrutura de persistência deve 

26 passar pela fachada `nucleus.database.database`. 

27 

28 Exemplo inválido: 

29 

30 orbits/database/persistence/persistence.py 

31 

32 from nucleus.database.drivers.driver_sqlalchemy import DriverSQLAlchemy 

33 

34 source = "orbits.database.persistence.persistence" 

35 target = "nucleus.database.drivers.driver_sqlalchemy" 

36 

37 Exemplo válido: 

38 

39 orbits/database/persistence/persistence.py 

40 

41 from nucleus.database.database import Database 

42 

43 source = "orbits.database.persistence.persistence" 

44 target = "nucleus.database.database" 

45 """ 

46 

47 # Prefixo que identifica os módulos de Orbits. 

48 ORBITS_PACKAGE = "orbits" 

49 

50 # Fachada pela qual a infraestrutura deve ser acessada. 

51 # 

52 # Usada SÓ na mensagem: a regra NÃO verifica que a fachada foi 

53 # usada — ela acusa quem importa o Driver. A distinção importa 

54 # para não se esperar dela mais do que ela faz. 

55 DATABASE_FACADE = "nucleus.database.database" 

56 

57 # Pacote do Driver, que não deve ser conhecido pelos Orbits. 

58 DRIVER_PACKAGE = "nucleus.database.drivers" 

59 

60 def code(self) -> str: 

61 return "SOLAR-ARCH-008" 

62 

63 def description(self) -> str: 

64 return ( 

65 "O Database do Nucleus é a fachada da " 

66 "infraestrutura de persistência." 

67 ) 

68 

69 def validate(self, dependency: Dependency) -> None: 

70 

71 # Toda a árvore `orbits.*` está sujeita: Planet, Database e 

72 # Satellites. O Driver só pode ser conhecido pelo próprio 

73 # Nucleus. 

74 if not self._is_orbits(dependency.source): 

75 return 

76 

77 if not self._is_driver(dependency.target): 

78 return 

79 

80 raise SolarException( 

81 self._build_message(dependency) 

82 ) 

83 

84 def _is_orbits(self, module: str) -> bool: 

85 

86 # Prefixo COM PONTO: `orbits` e `orbits.*` são Orbits; 

87 # `orbits_extra` não é. O ponto é o que separa os dois. 

88 return ( 

89 module == self.ORBITS_PACKAGE 

90 or module.startswith(f"{self.ORBITS_PACKAGE}.") 

91 ) 

92 

93 def _is_driver(self, target: str) -> bool: 

94 

95 # Mesmo cuidado: sem o ponto no `startswith`, um pacote 

96 # vizinho de nome parecido passaria como Driver. 

97 return ( 

98 target == self.DRIVER_PACKAGE 

99 or target.startswith(f"{self.DRIVER_PACKAGE}.") 

100 ) 

101 

102 def _build_message( 

103 self, 

104 dependency: Dependency 

105 ) -> str: 

106 

107 return ( 

108 f"{self.code()}: " 

109 f"O módulo de Orbits '{dependency.source}' não pode " 

110 f"acessar o Driver '{dependency.target}' diretamente. " 

111 f"O acesso deve passar pela fachada " 

112 f"'{self.DATABASE_FACADE}'. " 

113 f"Arquivo: {dependency.file}. " 

114 f"Linha: {dependency.line}." 

115 )

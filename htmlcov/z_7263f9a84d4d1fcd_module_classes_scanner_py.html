1# Scanner de CLASSES POR ARQUIVO. 

2# 

3# É o único scanner que pode não devolver nada para um arquivo: arquivo 

4# sem classe não gera contexto (ver `_scan_file`). Isso mantém o 

5# contexto alinhado com a pergunta da SOLAR-ARCH-032 — "este arquivo 

6# declara mais de um Model?" — que não faz sentido para arquivo sem 

7# classe. 

8import ast 

9from pathlib import Path 

10 

11from nucleus.rules.scanner.module_classes import ( 

12 DeclaredClass, 

13 ModuleClasses, 

14) 

15from nucleus.rules.scanner.source_files import ( 

16 resolve_exclusions, 

17 source_files, 

18) 

19 

20 

21class ModuleClassesScanner: 

22 """ 

23 Scanner responsável por descrever as CLASSES declaradas em cada 

24 arquivo do projeto. 

25 

26 Responde: 

27 

28 "Quais classes este arquivo declara, e em que linha?" 

29 

30 O scanner NÃO decide quantas são demais: isso é 

31 responsabilidade da Rule SOLAR-ARCH-032. 

32 

33 Ele descreve TODO arquivo que declara ao menos uma classe — 

34 inclusive os que têm uma só, e inclusive classes que não são 

35 Model. Assim o mesmo scanner serve a qualquer regra que fale 

36 sobre classes por arquivo. 

37 

38 Diretórios técnicos (cache, ambiente virtual, build) e 

39 diretórios de teste ficam de fora — ver 

40 nucleus.rules.scanner.source_files. 

41 

42 O código NÃO é executado: a análise é feita sobre a AST. 

43 """ 

44 

45 # Nome do módulo que marca um PACOTE: `__init__.py` não é um módulo 

46 # a mais, é o próprio diretório. 

47 PACKAGE_INIT = "__init__" 

48 

49 def __init__( 

50 self, 

51 root: Path, 

52 excluded_directories=None 

53 ): 

54 self.root = root.resolve() 

55 

56 # `resolve_exclusions` completa a lista com o padrão; passar uma 

57 # coleção VAZIA desliga a exclusão (é como um teste exercita um 

58 # arquivo dentro de `tests/`). 

59 self.excluded_directories = resolve_exclusions( 

60 excluded_directories 

61 ) 

62 

63 def scan(self) -> list[ModuleClasses]: 

64 """ 

65 Percorre o projeto e descreve os arquivos com classes. 

66 """ 

67 

68 modules: list[ModuleClasses] = [] 

69 

70 for file in source_files( 

71 self.root, 

72 self.excluded_directories 

73 ): 

74 

75 # `None` = arquivo sem classe. Só entra no resultado quem 

76 # tem o que descrever. 

77 module = self._scan_file(file) 

78 

79 if module is not None: 

80 modules.append(module) 

81 

82 return modules 

83 

84 def _scan_file(self, file: Path) -> ModuleClasses | None: 

85 """ 

86 Descreve as classes de um arquivo. 

87 

88 Devolve None quando o arquivo não declara classe alguma. 

89 """ 

90 

91 tree = ast.parse( 

92 file.read_text(encoding="utf-8"), 

93 filename=str(file) 

94 ) 

95 

96 # Ordem de LINHA, não de percurso da AST: classes aninhadas 

97 # apareceriam fora de ordem com ast.walk. 

98 declared = sorted( 

99 ( 

100 DeclaredClass(name=node.name, line=node.lineno) 

101 for node in ast.walk(tree) 

102 if isinstance(node, ast.ClassDef) 

103 ), 

104 key=lambda item: item.line 

105 ) 

106 

107 # Arquivo sem classe não vira contexto: não há pergunta a 

108 # responder sobre ele. 

109 if not declared: 

110 return None 

111 

112 # `as_posix()`: o relatório tem que sair igual no Windows e no 

113 # contêiner. 

114 relative = file.relative_to(self.root).as_posix() 

115 

116 return ModuleClasses( 

117 module=self._module_name(file), 

118 file=relative, 

119 classes=tuple(declared) 

120 ) 

121 

122 def _module_name(self, file: Path) -> str: 

123 """ 

124 Caminho pontuado do módulo. 

125 

126 orbits/database/models/user_model.py → 

127 "orbits.database.models.user_model" 

128 

129 orbits/database/models/__init__.py → 

130 "orbits.database.models" 

131 """ 

132 

133 # `list(...)` porque abaixo a lista é MODIFICADA — `parts` de um 

134 # `Path` é tupla. 

135 parts = list( 

136 file.relative_to(self.root).with_suffix("").parts 

137 ) 

138 

139 # `parts and` antes do `[-1]`: caminho vazio estouraria a 

140 # indexação. 

141 # 

142 # `__init__` é o marcador do pacote: o módulo é o próprio 

143 # pacote. 

144 if parts and parts[-1] == self.PACKAGE_INIT: 

145 parts.pop() 

146 

147 return ".".join(parts)

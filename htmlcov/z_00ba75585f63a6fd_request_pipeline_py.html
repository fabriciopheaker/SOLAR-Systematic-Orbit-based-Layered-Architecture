11class RequestPipeline: 

12 """ 

13 Mecanismo (middleware) do Nucleus que garante que toda 

14 entrada HTTP que transporta dados externos passe por 

15 validação e sanitização antes de alcançar o Planet. 

16 

17 A Atmosphere injeta o Sanitizer e o Validator apropriados ao 

18 tipo de conteúdo em cada HttpInput (injeção de dependência). 

19 O pipeline apenas aplica a regra SOLAR-ARCH-013 e executa as 

20 etapas na ordem correta: 

21 

22 HTTP Input → Validator → Sanitizer → Planet 

23 

24 A ordem NÃO é detalhe. A validação confere se os campos 

25 esperados EXISTEM; a sanitização limpa os valores. Invertendo 

26 isso, o Sanitizer tentaria limpar campo que pode não ter 

27 chegado — e quebraria antes de a validação poder dizer o que 

28 faltou. 

29 

30 Entradas que NÃO transportam dados externos passam direto, 

31 sem validação ou sanitização. 

32 """ 

33 

34 def __init__(self) -> None: 

35 

36 # Um motor com UMA regra: o contexto do pipeline é o HttpInput, e 

37 # misturar regras de outro contexto aqui quebraria a execução. 

38 self._engine = RuleEngine([ 

39 HttpInputMustPassThroughSanitizationAndValidation() 

40 ]) 

41 

42 def handle( 

43 self, 

44 http_input: HttpInput, 

45 payload 

46 ): 

47 """ 

48 Processa o payload de uma entrada HTTP. 

49 

50 Quando a entrada transporta dados externos: 

51 1. Aplica a regra SOLAR-ARCH-013; 

52 2. Valida o payload (os campos esperados existem?); 

53 3. Sanitiza o payload já validado; 

54 4. Retorna o payload sanitizado. 

55 

56 Caso contrário, retorna o payload original. 

57 """ 

58 

59 # Entrada sem dados externos passa direto — é o caso do 

60 # `/solar/health`: não há o que validar nem o que limpar. 

61 if not http_input.transports_data(): 

62 return payload 

63 

64 # Garante o contrato da SOLAR-ARCH-013. 

65 self._engine.validate(http_input) 

66 

67 # Os dois resolvidos antes de usar, e resolvidos com `isinstance` 

68 # (ver `_validator`/`_sanitizer`): na análise estática o campo pode 

69 # ser o NOME da classe, e aqui não dá para chamar método de string. 

70 validator = self._validator(http_input) 

71 sanitizer = self._sanitizer(http_input) 

72 

73 # VALIDAR e só então SANITIZAR — a ordem é a razão de existir 

74 # deste método (ver o docstring da classe). 

75 validator.validate(payload) 

76 

77 # Retorna o resultado do Sanitizer, e não o payload: o que segue 

78 # para o Planet é o dado já normalizado. 

79 return sanitizer.sanitize(payload) 

80 

81 def _sanitizer(self, http_input: HttpInput) -> Sanitizer: 

82 """ 

83 O pipeline roda em TEMPO DE EXECUÇÃO: exige a 

84 implementação real, não o nome declarado usado na 

85 validação estática. 

86 """ 

87 

88 # `isinstance` e não `is None`: uma STRING aqui (o nome que a 

89 # análise estática encontrou) passaria pela checagem de ausência 

90 # e quebraria depois, com erro obscuro. O `isinstance` recusa as 

91 # duas coisas de uma vez, com código de contrato. 

92 if not isinstance(http_input.sanitizer, Sanitizer): 

93 raise ContractException( 

94 "O Sanitizer fornecido não implementa o contrato " 

95 "Sanitizer.", 

96 code="INVALID_SANITIZER" 

97 ) 

98 

99 return http_input.sanitizer 

100 

101 def _validator(self, http_input: HttpInput) -> Validator: 

102 """ 

103 O pipeline roda em TEMPO DE EXECUÇÃO: exige a 

104 implementação real, não o nome declarado usado na 

105 validação estática. 

106 """ 

107 

108 # Mesmo cuidado do `_sanitizer`, e a mesma mensagem trocada: 

109 # validar o contrato ERRADO daria um diagnóstico enganoso. 

110 if not isinstance(http_input.validator, Validator): 

111 raise ContractException( 

112 "O Validator fornecido não implementa o contrato " 

113 "Validator.", 

114 code="INVALID_VALIDATOR" 

115 ) 

116

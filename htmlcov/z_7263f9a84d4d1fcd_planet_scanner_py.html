1# Scanner de Planet: quais existem e o que há na raiz de cada um. 

2# 

3# Este NÃO usa `source_files`: ali o que interessa são ARQUIVOS de 

4# código, e aqui são DIRETÓRIOS. O que ele reusa daquele módulo é só a 

5# decisão de "isto está excluído?", para um Planet de fixture não contar 

6# como Planet. 

7from pathlib import Path 

8 

9from nucleus.rules.scanner.planet_structure import PlanetStructure 

10from nucleus.rules.scanner.source_files import ( 

11 is_inside_excluded, 

12 resolve_exclusions, 

13) 

14 

15 

16class PlanetScanner: 

17 """ 

18 Scanner responsável por identificar os Planets da aplicação 

19 e descrever a estrutura da raiz de cada um. 

20 

21 Um Planet é um diretório posicionado imediatamente abaixo de 

22 um segmento "planets": 

23 

24 planets/user/ 

25 orbits/planets/user/ 

26 

27 O scanner responde apenas: 

28 

29 "Quais são os Planets e o que existe na raiz de cada um?" 

30 

31 A validação das regras arquiteturais é responsabilidade do 

32 RuleEngine e das Rules. 

33 """ 

34 

35 # Segmento que identifica a coleção de Planets 

36 # dentro do caminho de um diretório. 

37 PLANETS_SEGMENT = "planets" 

38 

39 # Diretórios que nunca descrevem a estrutura do Planet. 

40 # 

41 # Só `__pycache__`: ele nasce de uma execução e apareceria como 

42 # "peça extra" de todo Planet. Os outros nomes técnicos já foram 

43 # barrados antes, por `is_inside_excluded`. 

44 IGNORED_DIRECTORIES = frozenset({ 

45 "__pycache__", 

46 }) 

47 

48 def __init__(self, root: Path): 

49 """ 

50 Inicializa o Scanner. 

51 

52 Args: 

53 root: 

54 Diretório raiz do projeto analisado. 

55 """ 

56 

57 self.root = root.resolve() 

58 

59 def scan(self) -> list[PlanetStructure]: 

60 """ 

61 Percorre o projeto e descreve cada Planet encontrado. 

62 """ 

63 

64 planets: list[PlanetStructure] = [] 

65 

66 # `rglob("*")` traz arquivos E diretórios — só os diretórios 

67 # interessam, porque o que se descreve é a RAIZ do Planet. 

68 # `sorted` para a ordem ser estável entre execuções. 

69 for path in sorted(self.root.rglob("*")): 

70 

71 if not path.is_dir(): 

72 continue 

73 

74 # Teste e diretório técnico não descrevem a arquitetura: 

75 # um Planet de fixture não é um Planet. 

76 if is_inside_excluded(path, self.root): 

77 continue 

78 

79 # `is_inside_excluded` olha só os diretórios ACIMA do 

80 # caminho. O PRÓPRIO candidato precisa ser checado aqui — 

81 # sem isso, `orbits/planets/__pycache__` seria lido como 

82 # um Planet. 

83 if path.name in resolve_exclusions(): 

84 continue 

85 

86 # O critério de "é um Planet": estar DIRETAMENTE abaixo de 

87 # um segmento `planets`. É o `parent` que decide, e não o 

88 # caminho inteiro — por isso o mesmo teste vale para os dois 

89 # layouts conhecidos. 

90 if path.parent.name != self.PLANETS_SEGMENT: 

91 continue 

92 

93 planets.append( 

94 self._describe(path) 

95 ) 

96 

97 return planets 

98 

99 def _describe(self, path: Path) -> PlanetStructure: 

100 """ 

101 Descreve a raiz de um Planet: arquivos e diretórios. 

102 """ 

103 

104 files: set[str] = set() 

105 directories: set[str] = set() 

106 

107 # Só a RAIZ do Planet (`iterdir`, e não `walk`): a regra 019 

108 # exige peças no primeiro nível, e descer descreveria outra 

109 # coisa. 

110 for entry in path.iterdir(): 

111 

112 if entry.name in self.IGNORED_DIRECTORIES: 

113 continue 

114 

115 # Arquivo e diretório vão para conjuntos SEPARADOS: é essa 

116 # separação que faz um `repository.py` solto não satisfazer 

117 # o diretório `repository/`. 

118 if entry.is_dir(): 

119 directories.add(entry.name) 

120 else: 

121 files.add(entry.name) 

122 

123 return PlanetStructure( 

124 name=path.name, 

125 path=path.relative_to(self.root).as_posix(), 

126 files=frozenset(files), 

127 directories=frozenset(directories) 

128 )

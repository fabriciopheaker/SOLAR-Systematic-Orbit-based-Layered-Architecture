1# Resposta de erro segura + gancho para observadores. 

2# 

3# É a peça que fecha a SOLAR-ARCH-028 pelo lado da aplicação: o detalhe 

4# interno vai para o log (via `detail`) e o cliente recebe código, 

5# mensagem genérica e referência. 

6# 

7# O gancho `observers` é o único ponto de contato com o Observatório — e 

8# o contato é INVERTIDO: o Nucleus não importa `observatory`; o 

9# Observatório se pluga aqui. A direção da dependência é 

10# Observatory -> Nucleus. 

11import secrets 

12from collections.abc import Mapping 

13 

14 

15class ErrorHandler: 

16 """ 

17 Monta a resposta de erro SEGURA e separa o que NÃO pode sair 

18 da aplicação. 

19 

20 O problema que isto resolve (SOLAR-ARCH-028): a resposta para 

21 quem chamou precisa dizer O QUE deu errado, não COMO a 

22 aplicação é feita por dentro. 

23 

24 except Exception as error: 

25 logger.exception("falha ao processar") ← detalhe aqui 

26 return ErrorHandler().handle(error) ← resposta aqui 

27 

28 `handle()` devolve apenas: 

29 

30 { 

31 "code": "INVALID_UPLOAD" | "INTERNAL_ERROR", 

32 "message": mensagem GENÉRICA, 

33 "reference": identificador do ocorrido, 

34 } 

35 

36 A `reference` é o que liga a resposta ao log: quem reporta o 

37 problema informa o código, e o time encontra o detalhe no 

38 servidor — sem que o detalhe tenha saído de lá. 

39 

40 `detail()` monta o texto do LOG (aqui o detalhe interno é 

41 bem-vindo). 

42 

43 O gancho `observers` é por onde o Observatório se pluga: 

44 cada observador é chamado com (erro, payload, contexto) e 

45 recebe tudo o que precisa para registrar. O Nucleus NÃO 

46 conhece o Observatório — a direção da dependência é 

47 Observatory → Nucleus. 

48 

49 `expose_details=True` existe para o desenvolvimento local. 

50 Em produção isso é exatamente o vazamento que a regra proíbe. 

51 """ 

52 

53 # A mensagem é CONSTANTE, e não montada com dado do erro: qualquer 

54 # informação variável aqui seria o começo do vazamento. 

55 GENERIC_MESSAGE = "Não foi possível processar a solicitação." 

56 

57 # Código usado quando a exceção não traz o seu. 

58 INTERNAL_CODE = "INTERNAL_ERROR" 

59 

60 # Nome do atributo gravado NA EXCEÇÃO. O `solar_` na frente evita 

61 # colidir com atributo de exceção de biblioteca. 

62 REFERENCE_ATTRIBUTE = "solar_reference" 

63 

64 def __init__( 

65 self, 

66 expose_details: bool = False, 

67 identify=None, 

68 observers=None 

69 ): 

70 self.expose_details = expose_details 

71 

72 # Gerador de referência INJETÁVEL: o teste usa um previsível para 

73 # poder afirmar o conteúdo do payload. 

74 self._identify = identify or self._random_reference 

75 

76 # Tupla e não lista: os observadores não mudam por acidente no 

77 # meio de uma requisição — quem liga mais um usa `observe()`. 

78 self.observers = tuple(observers or ()) 

79 

80 def observe(self, *observers) -> None: 

81 """ 

82 Liga observadores ao handler. 

83 

84 Cada observador é um callable (erro, payload, contexto). O 

85 Nucleus não conhece o Observatório: conhece este gancho. 

86 """ 

87 

88 self.observers = (*self.observers, *observers) 

89 

90 # `secrets` e não `random`: a referência é lida por quem reporta o 

91 # problema e procurada no log — previsível, ela poderia ser 

92 # adivinhada para consultar o registro de outra pessoa. 

93 @staticmethod 

94 def _random_reference() -> str: 

95 return secrets.token_hex(6) 

96 

97 def handle( 

98 self, 

99 error: BaseException, 

100 context: Mapping | None = None 

101 ) -> dict: 

102 """ 

103 Payload SEGURO para quem chamou. 

104 

105 Antes de devolver, avisa os observadores ligados — é por 

106 aqui que o Observatório recebe o erro, sem que o Nucleus 

107 precise conhecê-lo. 

108 """ 

109 

110 # Três campos, e NENHUM deles é o texto da exceção: o payload é o 

111 # contrato com quem chamou. 

112 payload = { 

113 "code": self.code(error), 

114 "message": self.message(error), 

115 "reference": self.reference(error), 

116 } 

117 

118 # Avisa DEPOIS de montar o payload: o observador recebe o que foi 

119 # RESPONDIDO, e é essa ligação que permite achar o detalhe no log 

120 # a partir do que o cliente reportou. 

121 self._notify(error, payload, context) 

122 

123 return payload 

124 

125 def _notify( 

126 self, 

127 error: BaseException, 

128 payload: Mapping, 

129 context: Mapping | None 

130 ) -> None: 

131 """ 

132 Avisa cada observador ligado ao handler. 

133 

134 Observar é BEST-EFFORT: a falha de um observador não pode 

135 mudar a resposta de quem já recebeu um erro. 

136 """ 

137 

138 for observer in self.observers: 

139 

140 # `context or {}` para o observador não precisar tratar 

141 # `None`. 

142 try: 

143 observer(error, payload, context or {}) 

144 

145 # `except Exception: continue` é BEST-EFFORT deliberado: quem 

146 # já recebeu um erro não pode receber um segundo porque o 

147 # registrador falhou. O Observatório grava em SQLite, e o 

148 # banco pode estar indisponível. 

149 except Exception: 

150 continue 

151 

152 def code(self, error: BaseException) -> str: 

153 """ 

154 Código do erro quando a exceção traz um; senão, genérico. 

155 """ 

156 

157 # `getattr` com padrão: nem toda exceção tem `code`. E o `or` 

158 # cobre o `code` VAZIO, que também não serve como resposta. 

159 return getattr(error, "code", None) or self.INTERNAL_CODE 

160 

161 def message(self, error: BaseException) -> str: 

162 """ 

163 Mensagem para quem chamou. 

164 

165 Sempre genérica — exceto em desenvolvimento, quando 

166 `expose_details` foi ligado de propósito. 

167 """ 

168 

169 # O caminho NORMAL: mensagem fixa, sem nada do erro. 

170 if not self.expose_details: 

171 return self.GENERIC_MESSAGE 

172 

173 # Só com `expose_details` ligado, que é para desenvolvimento 

174 # local. Em produção isto é exatamente o vazamento que a regra 

175 # proíbe. 

176 return ( 

177 f"{self.GENERIC_MESSAGE} " 

178 f"({type(error).__name__}: {error})" 

179 ) 

180 

181 def reference(self, error: BaseException) -> str: 

182 """ 

183 Identificador estável do ocorrido. 

184 

185 Fica guardado na PRÓPRIA exceção, para que `handle()` e 

186 `detail()` produzam sempre a mesma referência. 

187 """ 

188 

189 # Já existe? Devolve a MESMA: `handle()` e `detail()` precisam 

190 # produzir a mesma referência, senão a busca no log não acha 

191 # nada. 

192 reference = getattr(error, self.REFERENCE_ATTRIBUTE, None) 

193 

194 if reference is not None: 

195 return reference 

196 

197 reference = self._identify() 

198 

199 # Grava na PRÓPRIA exceção — é o que faz a segunda chamada cair 

200 # no `if` acima. 

201 try: 

202 setattr(error, self.REFERENCE_ATTRIBUTE, reference) 

203 except AttributeError: 

204 # Exceções com __slots__ não aceitam atributo novo; a 

205 # referência continua válida, só não fica guardada. 

206 pass 

207 

208 return reference 

209 

210 def detail(self, error: BaseException) -> str: 

211 """ 

212 Texto para o LOG do servidor. 

213 """ 

214 

215 # Aqui o detalhe interno é BEM-VINDO: este texto vai para o log do 

216 # servidor, e é ele que dá sentido à referência. A referência vem 

217 # entre colchetes para ser fácil de procurar no log. 

218 return ( 

219 f"[{self.reference(error)}] " 

220 f"{type(error).__name__}: {error}" 

221 )

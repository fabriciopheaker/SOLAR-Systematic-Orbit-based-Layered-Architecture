1# SOLAR-ARCH-023 — segredo reconhecido pelo CONTEÚDO. 

2# 

3# Irmã da SOLAR-ARCH-020, e a diferença é o que cada uma olha: 

4# 

5# 020 o NOME da variável (SECRET_KEY, password, ...) 

6# 023 o VALOR — pega o segredo mesmo com nome inofensivo 

7# (CHAVE = "AKIA...") 

8# 

9# As duas podem apontar para o mesmo trecho. São regras separadas 

10# porque a 020 não enxerga segredo de nome neutro, e a 023 não 

11# enxerga segredo que não case com nenhum padrão conhecido. 

12from nucleus.exceptions.solar_exception import SolarException 

13from nucleus.rules.rule import Rule 

14from nucleus.rules.scanner.secret_occurrence import SecretOccurrence 

15 

16 

17class SecretsMustNotAppearInSource(Rule): 

18 """ 

19 Regra arquitetural: 

20 

21 SOLAR-ARCH-023 

22 

23 Secrets NÃO podem estar no código-fonte. 

24 

25 Diferente da SOLAR-ARCH-020 — que detecta pelo NOME da 

26 variável (SECRET_KEY, password, ...) — esta regra detecta pelo 

27 CONTEÚDO: o segredo é reconhecido mesmo que o nome seja 

28 inofensivo. 

29 

30 É VIOLAÇÃO: 

31 

32 CHAVE = "AKIAIOSFODNN7EXAMPLE" ❌ 

33 HEADER = "-----BEGIN RSA PRIVATE KEY-----" ❌ 

34 TOKEN = "ghp_0123456789012345678901234567" ❌ 

35 JWT = "eyJhbGciOiJIUzI1NiJ9.eyJzdWIiOiIxIn0.x" ❌ 

36 

37 O segredo pertence ao Environment, e o código o obtém de lá: 

38 

39 CHAVE = os.environ["AWS_ACCESS_KEY_ID"] ✅ 

40 CHAVE = environment.get("AWS_ACCESS_KEY_ID") ✅ 

41 

42 Um casamento de padrão, porém, NÃO prova que existe segredo: o 

43 scanner também marca os locais que exigem contexto. Estes são 

44 ignorados, e cada um pode ser desligado no construtor: 

45 

46 documentation docstring/texto solto — não é valor usado 

47 pelo programa (ignore_documentation); 

48 example valor de exemplo/placeholder conhecido 

49 (ignore_examples); 

50 suppressed trecho anotado com "solar:allow-secret" 

51 (ignore_suppressed). 

52 

53 A exceção fica VISÍVEL no código e é revisável no pull 

54 request — nenhuma delas é um ignore escondido em arquivo de 

55 configuração. 

56 

57 O valor do segredo NÃO é registrado na mensagem: reportar o 

58 valor transformaria o relatório em um novo vazamento. 

59 

60 O contexto desta regra é um SecretOccurrence. 

61 """ 

62 

63 def __init__( 

64 self, 

65 ignore_documentation: bool = True, 

66 ignore_examples: bool = True, 

67 ignore_suppressed: bool = True 

68 ): 

69 # Desligáveis porque cada uma troca rigor por ruído: com 

70 # `ignore_examples=False`, todo placeholder do próprio 

71 # manual viraria violação. 

72 self.ignore_documentation = ignore_documentation 

73 self.ignore_examples = ignore_examples 

74 self.ignore_suppressed = ignore_suppressed 

75 

76 def code(self) -> str: 

77 return "SOLAR-ARCH-023" 

78 

79 def description(self) -> str: 

80 return "Secrets não podem estar no código-fonte." 

81 

82 def validate(self, occurrence: SecretOccurrence) -> None: 

83 

84 # A justificativa é avaliada AQUI, e não no scanner: o 

85 # scanner DESCREVE ("é docstring", "é exemplo"); decidir o 

86 # que fazer com essa informação é política, e política é da 

87 # regra. 

88 if self._is_justified(occurrence): 

89 return 

90 

91 raise SolarException( 

92 self._build_message(occurrence) 

93 ) 

94 

95 def _is_justified(self, occurrence: SecretOccurrence) -> bool: 

96 """ 

97 Indica se a ocorrência tem justificativa para ser ignorada. 

98 """ 

99 

100 # `or` encadeado: basta UMA justificativa. E cada termo 

101 # começa pela flag, porque com ela desligada a 

102 # justificativa não vale — é assim que "não ignorar 

103 # exemplos" transforma exemplo em violação. 

104 return ( 

105 self.ignore_documentation and occurrence.documentation 

106 ) or ( 

107 self.ignore_examples and occurrence.example 

108 ) or ( 

109 self.ignore_suppressed and occurrence.suppressed 

110 ) 

111 

112 def _build_message( 

113 self, 

114 occurrence: SecretOccurrence 

115 ) -> str: 

116 

117 # O local e o TIPO do padrão, nunca o valor: mensagem com o 

118 # segredo dentro viraria um novo vazamento — no terminal, 

119 # no log e no arquivo de auditoria. 

120 return ( 

121 f"{self.code()}: " 

122 f"segredo do tipo '{occurrence.pattern}' encontrado em " 

123 f"{occurrence.file}:{occurrence.line}. " 

124 f"Secrets devem vir do Environment e nunca do " 

125 f"código-fonte. (o valor não é exibido por segurança)" 

126 )

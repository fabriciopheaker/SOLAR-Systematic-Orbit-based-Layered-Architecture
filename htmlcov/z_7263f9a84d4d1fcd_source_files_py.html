1""" 

2Quais arquivos são CÓDIGO DO PROJETO. 

3 

4Nem todo `.py` sob a raiz participa da arquitetura. 

5 

6Os TESTES são o caso que motiva este módulo: para validar um 

7comportamento e descobrir falhas, um teste precisa justamente 

8quebrar as regras — importar a ORM direto, montar um SQL na mão, 

9declarar um Model na hora. Cobrar arquitetura de teste é cobrar 

10do lugar errado, e o resultado é a regra ficar vermelha por causa 

11de um `assert`. 

12 

13Por isso a exclusão acontece no SCANNER, não na Rule: o arquivo de 

14teste nem chega a ser interpretado. 

15 

16Diretórios técnicos (cache, ambiente virtual, build) saem pelo 

17mesmo caminho, e por um motivo mais simples: não são código 

18escrito à mão. 

19 

20O escopo é por DIRETÓRIO, não por nome de arquivo: um `tests.py` 

21continua sendo analisado — é um arquivo como outro qualquer. A 

22exceção são os nomes que SÃO infraestrutura de teste em qualquer 

23lugar (`conftest.py`, `test_*.py`, `*_test.py`). 

24""" 

25 

26from collections.abc import Iterable 

27from pathlib import Path 

28 

29 

30# Diretórios que NÃO são código do projeto. 

31DEFAULT_EXCLUDED_DIRECTORIES = frozenset({ 

32 # Técnicos: cache, ambiente virtual, artefato de build. 

33 ".git", 

34 ".mypy_cache", 

35 ".pytest_cache", 

36 ".ruff_cache", 

37 ".tox", 

38 ".venv", 

39 "__pycache__", 

40 "build", 

41 "dist", 

42 "node_modules", 

43 "venv", 

44 # Testes e material de apoio: quebrar a regra aqui é legítimo. 

45 "examples", 

46 "fixtures", 

47 "samples", 

48 "spec", 

49 "specs", 

50 "test", 

51 "tests", 

52}) 

53 

54# Nomes que são infraestrutura de teste ONDE QUER QUE ESTEJAM. 

55# 

56# Três formatos porque não existe um só: `conftest.py` (nome do 

57# pytest), `test_*.py` (prefixo, o usado neste projeto) e `*_test.py` 

58# (sufixo, comum em outros). Sem os três, um teste fora de `tests/` 

59# voltaria a ser cobrado pela arquitetura. 

60TEST_FILENAMES = frozenset({ 

61 "conftest.py", 

62}) 

63 

64TEST_FILENAME_PREFIXES = ( 

65 "test_", 

66) 

67 

68TEST_FILENAME_SUFFIXES = ( 

69 "_test.py", 

70)

73def source_files( 

74 root: Path, 

75 excluded_directories: Iterable[str] | None = None 

76) -> list[Path]: 

77 """ 

78 Arquivos `.py` que fazem parte do projeto, em ordem. 

79 """ 

80 

81 # `resolve()` UMA vez, e a MESMA raiz passada adiante: se cada 

82 # chamada resolvesse de novo, `relative_to` poderia comparar 

83 # caminhos escritos de formas diferentes no Windows. 

84 resolved = Path(root).resolve() 

85 

86 # `sorted` no `rglob`: a ORDEM tem que ser estável. Sem ela, dois 

87 # arquivos poderiam alternar de posição entre execuções e o 

88 # relatório sairia diferente a cada rodada. 

89 return [ 

90 file 

91 for file in sorted(resolved.rglob("*.py")) 

92 if belongs_to_project(file, resolved, excluded_directories) 

93 ] 

94 

95 

96def belongs_to_project( 

97 file: Path, 

98 root: Path, 

99 excluded_directories: Iterable[str] | None = None 

100) -> bool: 

101 """ 

102 Indica se o arquivo participa da arquitetura. 

103 """ 

104 

105 # Nome primeiro: é a checagem mais barata e a que vale em qualquer 

106 # lugar do projeto. 

107 if is_test_file(file.name): 

108 return False 

109 

110 return not is_inside_excluded( 

111 file, 

112 root, 

113 excluded_directories 

114 )

117def is_test_file(name: str) -> bool: 

118 """ 

119 Indica se o nome é de um arquivo de teste. 

120 """ 

121 

122 # Os três em `or`: qualquer um basta. `startswith`/`endswith` 

123 # aceitam TUPLA, então os formatos ficam em constantes e ampliar é 

124 # editar a lista. 

125 return ( 

126 name in TEST_FILENAMES 

127 or name.startswith(TEST_FILENAME_PREFIXES) 

128 or name.endswith(TEST_FILENAME_SUFFIXES) 

129 )

132def is_inside_excluded( 

133 path: Path, 

134 root: Path, 

135 excluded_directories: Iterable[str] | None = None 

136) -> bool: 

137 """ 

138 Indica se o caminho está sob um diretório excluído. 

139 

140 Olha só os DIRETÓRIOS entre a raiz e o próprio caminho. 

141 """ 

142 

143 # Resolvido AQUI e não recebido pronto: a função é pública e pode 

144 # ser chamada direto — o `PlanetScanner` faz exatamente isso. 

145 excluded = resolve_exclusions(excluded_directories) 

146 

147 # `parts[:-1]` — só os DIRETÓRIOS do caminho. O último elemento é o 

148 # próprio arquivo, e um arquivo chamado `tests.py` não é o 

149 # diretório `tests`. 

150 return any( 

151 directory in excluded 

152 for directory in Path(path).relative_to(root).parts[:-1] 

153 ) 

154 

155 

156def resolve_exclusions( 

157 excluded_directories: Iterable[str] | None = None 

158) -> frozenset[str]: 

159 """ 

160 Exclusões em vigor. 

161 

162 `None` usa o padrão; uma coleção (mesmo vazia) substitui o 

163 padrão — é assim que um scanner permite desligar a exclusão. 

164 """ 

165 

166 # `is None` e não `if not ...`: coleção VAZIA é uma escolha ("não 

167 # exclua nada"). Tratá-la como "sem argumento" tiraria a 

168 # possibilidade de um scanner varrer tudo. 

169 if excluded_directories is None: 

170 return DEFAULT_EXCLUDED_DIRECTORIES 

171 

172 return frozenset(excluded_directories)

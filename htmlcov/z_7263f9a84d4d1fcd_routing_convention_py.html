6import ast 

7from dataclasses import dataclass

10@dataclass(frozen=True) 

11class RoutingConvention: 

12 """ 

13 Descreve COMO um framework declara rotas. 

14 

15 A SOLAR não deve ficar presa a um framework específico. 

16 Cada framework fornece sua própria RoutingConvention, e o 

17 RouteScanner usa a convenção para reconhecer as declarações. 

18 

19 route_decorators: 

20 Nomes de decorators que declaram rotas 

21 (ex.: "route" para @app.route). 

22 

23 route_registrations: 

24 Nomes de funções que registram rotas explicitamente 

25 (ex.: "add_url_rule"). 

26 

27 Dessa forma, adicionar suporte a outro framework não exige 

28 alterar a Rule: basta fornecer outra RoutingConvention. 

29 """ 

30 

31 name: str 

32 

33 # Dois vocabulários porque há DUAS formas de declarar rota: o 

34 # decorator (`@bp.route`) e o registro por chamada 

35 # (`bp.add_url_rule`). A SOLAR-ARCH-019 aceita qualquer uma 

36 # delas. 

37 route_decorators: frozenset[str] 

38 route_registrations: frozenset[str]

41def is_super_delegation(node: ast.Call) -> bool: 

42 """ 

43 Indica se a chamada é `super().algo(...)`. 

44 

45 É a chamada da BASE — quem ESTENDE o framework delegando a ele —, e 

46 não a declaração de uma rota no código da aplicação: 

47 

48 class SolarBlueprint(Blueprint): 

49 def add_url_rule(self, ...): 

50 ... 

51 super().add_url_rule(...) ← delegação, não rota 

52 

53 Sem esta distinção, o mecanismo que existe para deixar a ROTA 

54 declarar os guardas seria acusado pelas regras de rota — que é 

55 exatamente o contrário do que ele serve. 

56 """ 

57 

58 if not isinstance(node.func, ast.Attribute): 

59 return False 

60 

61 value = node.func.value 

62 

63 if not isinstance(value, ast.Call): 

64 return False 

65 

66 return ( 

67 isinstance(value.func, ast.Name) 

68 and value.func.id == "super" 

69 )

72def flask_routing_convention() -> RoutingConvention: 

73 """ 

74 Convenção de roteamento do Flask. 

75 

76 Decorators: 

77 @app.route(...) 

78 @blueprint.route(...) 

79 @api.route(...) 

80 @router.get(...) / @router.post(...) / etc. 

81 

82 Registro explícito: 

83 blueprint.add_url_rule(...) 

84 """ 

85 

86 # `get`/`post`/... entram junto de `route` porque um `MethodView` 

87 # ou um Router de extensão declara rota pelo VERBO 

88 # (`@router.get("/x")`) — e isso é declaração de rota, mesmo sem 

89 # a palavra "route" no código. 

90 return RoutingConvention( 

91 name="flask", 

92 route_decorators=frozenset({ 

93 "route", 

94 "get", 

95 "post", 

96 "put", 

97 "patch", 

98 "delete", 

99 "head", 

100 "options", 

101 }), 

102 # O registro explícito é o caso de `routes.py`, onde a rota 

103 # é montada por chamada e não por decorator. 

104 route_registrations=frozenset({ 

105 "add_url_rule", 

106 }), 

107 )

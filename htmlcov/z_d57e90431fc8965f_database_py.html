1""" 

2Fachada de persistência da aplicação. 

3 

4A aplicação — Orbits, Planets — fala SEMPRE com `Database` e 

5NUNCA com o driver (SOLAR-ARCH-008). Trocar SQLAlchemy por outra 

6tecnologia é trocar o driver injetado em `init()`, sem tocar em 

7quem consulta. 

8""" 

9 

10import importlib 

11import pkgutil 

12 

13from .drivers.interface.database_driver import DatabaseDriver 

14from nucleus.exceptions.contract_exception import ContractException 

15from nucleus.exceptions.database_exception import DatabaseException 

16 

17class Database: 

18 

19 

20 # O driver ligado em `init()`. Enquanto for None, a aplicação 

21 # ainda não pode consultar nada. 

22 _driver: DatabaseDriver | None = None 

23 

24 # Publicados a partir do driver logo depois da conexão. É por 

25 # eles que um Model usa a ORM SEM importar sqlalchemy — o que 

26 # mantém a SOLAR-ARCH-010 de pé: só o driver conhece a 

27 # tecnologia. 

28 orm = None 

29 model = None 

30 

31 # De ONDE os Models sao importados. 

32 # 

33 # E a unica coisa que a fachada sabe sobre o CONTEUDO do projeto — 

34 # e ela sabe porque e ela quem monta o banco: a ORM so conhece o 

35 # Model que ja passou pelo import, e quem importa e ela. 

36 MODELS_PACKAGE = "orbits.database.models" 

37 

38 # MONTA as estruturas do banco ao iniciar. E o BOTAO. 

39 # 

40 # ATENCAO: em PRODUCAO isto deve ficar DESLIGADO. Sincronizar 

41 # esquema cria, altera e REMOVE estrutura — e um engano aqui custa 

42 # dado. Em producao o esquema pertence ao DEPLOY, nao a subida. 

43 # 

44 # E tambem o ponto de troca para quem quiser migrations: com 

45 # `SYNC_ON_START = False`, quem cuida do esquema passa a ser a 

46 # ferramenta de migracao, rodada antes de subir a aplicacao. 

47 SYNC_ON_START = True 

48 

49 @classmethod 

50 def init(cls, driver: DatabaseDriver) -> None: 

51 """ 

52 Inicializa o mecanismo de persistência da aplicação. 

53 O driver recebido deve implementar o contrato DatabaseDriver. 

54 """ 

55 

56 

57 # Recusa driver fora do contrato na MONTAGEM, e não na 

58 # primeira consulta em produção. 

59 if not isinstance(driver, DatabaseDriver): 

60 raise ContractException( 

61 "O driver fornecido não implementa o contrato DatabaseDriver.", 

62 code="INVALID_DATABASE_DRIVER" 

63 ) 

64 

65 cls._driver = driver 

66 

67 # A fachada não escolhe a tecnologia: ela repassa o que o 

68 # driver oferece. 

69 cls.orm = cls._driver.orm() 

70 cls.model = cls._driver.model() 

71 

72 # Conecta já na montagem. Subir com o banco inalcançável falha 

73 # aqui, com a causa à vista, em vez de falhar na requisição de 

74 # um usuário. 

75 cls._driver.connect() 

76 

77 # E MONTA AS ESTRUTURAS: o banco inicia montado, e não só 

78 # conectado. Banco conectado sem as tabelas certas é falha 

79 # adiada para a primeira consulta. 

80 # 

81 # A ordem importa: os Models só podem ser importados DEPOIS de 

82 # `orm` e `model` estarem publicados, porque o Model declara 

83 # `class Users(Database.orm)` no próprio import. 

84 if cls.SYNC_ON_START: 

85 cls.sync() 

86 

87 @classmethod 

88 def driver(cls) -> DatabaseDriver:

90 Retorna o driver atualmente configurado. 

91 """

94 # inicializar. O código estável permite separar os dois no log. 

95 if cls._driver is None: 

96 raise DatabaseException( 

97 "O Database ainda não foi inicializado.", 

98 code="DATABASE_NOT_INITIALIZED"

100 

101 return cls._driver

105 @classmethod 

106 def session(cls): 

107 """ 

108 Retorna a sessão de trabalho da ORM. 

109 

110 É por onde a Persistence lê e grava. Vem do driver: a fachada 

111 não cria sessão nenhuma, só repassa o que ele oferece — igual 

112 ao que já faz com o `orm` e o `model`.

114 

115 return cls.driver().session()

117 @classmethod 

118 def sync(cls, *, prune=True) -> list[str]: 

119 """ 

120 Monta o esquema do banco a partir dos Models. 

121 

122 Cria o que falta, altera o que mudou e — por padrão — REMOVE o 

123 que não existe mais no Model. 

124 

125 Quem sabe fazer isso é o driver. Antes de pedir, a fachada 

126 IMPORTA os Models: a ORM só conhece o que já passou pelo 

127 import, e um Model que ninguém importou é, para ela, uma 

128 tabela que não existe. 

129 

130 Devolve os módulos de Model que foram importados. 

131 """ 

132 

133 # Erro de MONTAGEM, não de uso. A checagem vem ANTES dos 

134 # imports: importar um Model sem ORM publicada estoura lá 

135 # dentro, sem explicar que faltou inicializar o banco. 

136 cls.driver() 

137 

138 imported = cls._import_models() 

139 

140 cls.driver().sync(prune=prune) 

141 

142 return imported

144 @classmethod 

145 def _import_models(cls) -> list[str]: 

146 """ 

147 Importa todos os Models do pacote e devolve os nomes. 

148 

149 Importar é o que REGISTRA a tabela na ORM: não há registro 

150 separado. Só o primeiro nível do pacote é varrido, o que basta 

151 porque a regra do projeto é um Model por ARQUIVO. 

152 

153 A ordem é ALFABÉTICA, e não a do sistema de arquivos: o 

154 resultado precisa ser o mesmo em qualquer máquina. 

155 """ 

156 

157 package = importlib.import_module(cls.MODELS_PACKAGE) 

158 

159 names = sorted( 

160 name for _, name, _ in pkgutil.iter_modules(package.__path__) 

161 ) 

162 

163 imported: list[str] = [] 

164 

165 for name in names: 

166 

167 module = importlib.import_module(f"{cls.MODELS_PACKAGE}.{name}") 

168 

169 imported.append(module.__name__) 

170 

171 return imported

174 def close(cls) -> None: 

175 """ 

176 Encerra a conexão com o mecanismo de persistência. 

177 """ 

178

185 cls._driver.disconnect()

188 # precisa injetar um driver novo. 

189 cls._driver = None

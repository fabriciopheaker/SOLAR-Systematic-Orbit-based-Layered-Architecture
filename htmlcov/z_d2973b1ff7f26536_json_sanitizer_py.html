1# Sanitizador de corpo JSON. 

2# 

3# É a guarda de ENTRADA (SOLAR-ARCH-013) que LIMPA: o 

4# `JsonValidator` confere o que existe, este normaliza o que 

5# chegou. A ordem é essa — conferir antes de limpar — e a razão 

6# está no contrato `Sanitizer`. 

7# 

8# Ele não julga nada: não decide se o e-mail é válido nem se o nome 

9# é aceitável. O que ele tira do dado é o que NÃO é dado — espaço em 

10# volta, valor nulo, chave sem conteúdo. 

11 

12from collections.abc import Mapping 

13 

14from nucleus.contracts.sanitizer import Sanitizer 

15from nucleus.exceptions.validation_exception import ValidationException 

16 

17 

18class JsonSanitizer(Sanitizer): 

19 """ 

20 Normaliza um corpo JSON antes de ele virar dado da aplicação. 

21 

22 {"name": " Ana ", "apelido": None} 

23 → {"name": "Ana"} 

24 

25 O que faz: 

26 

27 - recusa payload que não seja um objeto JSON; 

28 - tira espaço em volta de todo valor de TEXTO; 

29 - descarta a chave cujo valor é `None`. 

30 

31 O que NÃO faz: conferir campo obrigatório (é do `JsonValidator`), 

32 converter tipo e decidir regra de negócio (é do UseCase). 

33 

34 É ESPECÍFICO do tipo de conteúdo `application/json`, como manda o 

35 contrato: form data e arquivo têm o seu. 

36 """ 

37 

38 # Mesmo código do validador para o mesmo problema: o corpo não é 

39 # objeto JSON. Um código por guarda faria quem trata a resposta 

40 # ter que conhecer os dois. 

41 INVALID_PAYLOAD = "INVALID_JSON_BODY" 

42 

43 def sanitize(self, payload) -> dict: 

44 """ 

45 Devolve uma CÓPIA normalizada do corpo. 

46 """ 

47 

48 # `Mapping` e não `dict`: o corpo pode chegar como qualquer 

49 # mapeamento, e recusar isso seria recusar dado bom pelo tipo 

50 # da caixa em que ele veio. 

51 if not isinstance(payload, Mapping): 

52 

53 raise ValidationException( 

54 "O corpo da requisição deve ser um objeto JSON.", 

55 code=self.INVALID_PAYLOAD 

56 ) 

57 

58 # Dicionário NOVO, e não `payload` alterado no lugar: quem 

59 # chamou ainda tem a referência do corpo recebido, e mudá-lo 

60 # por baixo seria surpresa. 

61 sanitized: dict = {} 

62 

63 for key, value in payload.items(): 

64 

65 # Chave sem valor não vira campo. `None` é a ausência 

66 # escrita de outra forma, e deixá-la passar empurraria 

67 # para o UseCase um problema que é de entrada. 

68 if value is None: 

69 continue 

70 

71 # `strip()` só em TEXTO: número e booleano não têm espaço 

72 # em volta, e chamar `strip` neles estouraria. Espaço em 

73 # texto, porém, é ruído de digitação — " Ana " e "Ana" 

74 # são o mesmo nome. 

75 if isinstance(value, str): 

76 sanitized[key] = value.strip() 

77 continue 

78 

79 sanitized[key] = value 

80 

81 return sanitized

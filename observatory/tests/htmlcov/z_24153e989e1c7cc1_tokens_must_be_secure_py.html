1# SOLAR-ARCH-027 — as três exigências do token. 

2# 

3# Os problemas são independentes e a regra acumula todos, em vez de 

4# parar no primeiro: corrigir `algorithm` e só então descobrir que 

5# faltava `exp` obrigaria a rodar o check de novo. 

6# 

7# O critério é sempre "VISTO errado", nunca "não sei" — ver 

8# `_problems`, que só acusa `False` explícito (o contexto guarda 

9# `None` para "não foi possível ver"). 

10from nucleus.exceptions.solar_exception import SolarException 

11from nucleus.rules.rule import Rule 

12from nucleus.rules.scanner.token_usage import TokenUsage 

13 

14 

15class TokensMustBeSecure(Rule): 

16 """ 

17 Regra arquitetural: 

18 

19 SOLAR-ARCH-027 

20 

21 JWT/Token seguro: todo token deve ser 

22 

23 1. assinado com algoritmo EXPLÍCITO e seguro; 

24 2. VERIFICADO na leitura; 

25 3. emitido COM expiração. 

26 

27 É VIOLAÇÃO: 

28 

29 jwt.encode(payload, key, algorithm="none") ❌ 

30 jwt.encode(payload, key, algorithm="HS256") ❌ (sem "exp") 

31 jwt.decode(token, key) ❌ (algoritmo implícito) 

32 jwt.decode(token, key, 

33 options={"verify_signature": False}) ❌ 

34 

35 Correto: 

36 

37 jwt.encode( 

38 {"sub": user_id, "exp": expires_at}, 

39 key, 

40 algorithm="HS256", 

41 ) ✅ 

42 

43 jwt.decode(token, key, algorithms=["HS256"]) ✅ 

44 

45 A chave de assinatura deve vir do Environment — a 

46 SOLAR-ARCH-020/023 cuidam disso. 

47 

48 O contexto desta regra é um TokenUsage. 

49 """ 

50 

51 # Algoritmos que NÃO assinam o token. 

52 # 

53 # `"none"` é o algoritmo SEM ASSINATURA da própria especificação 

54 # do JWT — existe, e bibliotecas permissivas o aceitam; `""` 

55 # pega o caso de `algorithm=""`. A comparação é feita em 

56 # minúsculas por `strip().lower()` dentro de `_problems`. 

57 UNSAFE_ALGORITHMS = frozenset({ 

58 "none", 

59 "", 

60 }) 

61 

62 def code(self) -> str: 

63 return "SOLAR-ARCH-027" 

64 

65 def description(self) -> str: 

66 return ( 

67 "JWT/Token deve ser assinado com algoritmo explícito, " 

68 "ter expiração e ser verificado na leitura." 

69 ) 

70 

71 def validate(self, usage: TokenUsage) -> None: 

72 

73 problems = self._problems(usage) 

74 

75 if not problems: 

76 return 

77 

78 raise SolarException( 

79 self._build_message(usage, problems) 

80 ) 

81 

82 def _problems(self, usage: TokenUsage) -> list[str]: 

83 

84 problems: list[str] = [] 

85 

86 # `if`/`elif`: algoritmo declarado só pode ter UM problema — 

87 # ou veio e está vazio/inseguro, ou veio e está bom. Não há 

88 # caso em que as duas condições se apliquem. 

89 if usage.algorithm is None: 

90 problems.append( 

91 "algoritmo não declarado explicitamente" 

92 ) 

93 

94 # `strip().lower()` porque `"NONE"`, `" none "` e `"none"` 

95 # são o mesmo algoritmo. Comparar cru deixaria passar a 

96 # variação de caixa — que é justamente o jeito de driblar 

97 # uma lista de proibidos. 

98 elif usage.algorithm.strip().lower() in self.UNSAFE_ALGORITHMS: 

99 problems.append( 

100 f"algoritmo inseguro '{usage.algorithm}'" 

101 ) 

102 

103 # `is False` e não `not usage.verifies_signature`: `None` 

104 # significa "não foi possível ver", e acusar a dúvida daria 

105 # falso positivo em verificação montada em tempo de execução. 

106 if usage.verifies_signature is False: 

107 problems.append( 

108 "verificação de assinatura desativada" 

109 ) 

110 

111 # Só na EMISSÃO: um token que apenas é LIDO não tem por que 

112 # declarar `exp` — quem responde por isso é quem o emitiu. 

113 if usage.operation == "encode" and usage.has_expiration is False: 

114 problems.append("token emitido sem expiração ('exp')") 

115 

116 return problems 

117 

118 def _build_message( 

119 self, 

120 usage: TokenUsage, 

121 problems: list[str] 

122 ) -> str: 

123 

124 return ( 

125 f"{self.code()}: " 

126 f"uso inseguro de token em {usage.operation} " 

127 f"({usage.file}:{usage.line}): " 

128 f"{'; '.join(problems)}. " 

129 f"Tokens devem ser assinados com algoritmo explícito, " 

130 f"ter expiração e ser verificados na leitura." 

131 )

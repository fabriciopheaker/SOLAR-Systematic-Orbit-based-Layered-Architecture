1# SOLAR-ARCH-017 — a rota tem que desembocar em um Controller. 

2# 

3# É o par da SOLAR-ARCH-016: a 016 cuida de COMO a rota é declarada; 

4# esta cuida de PARA QUEM ela aponta. Juntas, garantem que o fluxo 

5# Orbit -> Planet Routes -> Controller exista de verdade. 

6from nucleus.exceptions.solar_exception import SolarException 

7from nucleus.rules.rule import Rule 

8from nucleus.rules.scanner.route_declaration import RouteDeclaration 

9 

10 

11class RoutesMustPointToController(Rule): 

12 """ 

13 Regra arquitetural: 

14 

15 SOLAR-ARCH-017 

16 

17 Toda rota deve apontar para um Controller. 

18 

19 Fluxo exigido: 

20 

21 Orbits → Planet Routes → Controller → Input → UseCase 

22 → Entity / Repository → Output → Resource → HTTP Response 

23 

24 É PROIBIDO uma rota apontar diretamente para: 

25 

26 - UseCase 

27 - Repository 

28 - Entity 

29 - Service 

30 - qualquer componente de domínio ou persistência 

31 - lambda / expressão inline 

32 

33 Exemplo proibido: 

34 

35 api.add_url_rule("/users", "create", CreateUserUseCase.execute) 

36 api.add_url_rule("/api", "home", lambda: jsonify({...})) 

37 

38 Exemplo correto: 

39 

40 api.add_url_rule("/users", "create", UserController.create) 

41 

42 ESCOPO: esta regra avalia apenas rotas registradas 

43 explicitamente (REGISTRATION). Rotas declaradas por decorator 

44 são tratadas pela SOLAR-ARCH-016. 

45 """ 

46 

47 # Sufixo que identifica um Controller da SOLAR. 

48 # 

49 # Mesma convenção da SOLAR-ARCH-022: o reconhecimento é pelo 

50 # NOME. É o que o scanner consegue ver sem executar o código — 

51 # descobrir a classe de verdade exigiria importar a aplicação. 

52 CONTROLLER_SUFFIX = "Controller" 

53 

54 def code(self) -> str: 

55 return "SOLAR-ARCH-017" 

56 

57 def description(self) -> str: 

58 return "Toda rota deve apontar para um Controller." 

59 

60 def validate(self, declaration: RouteDeclaration) -> None: 

61 

62 # Só o estilo REGISTRATION: a rota por decorator nem chega a 

63 # ser avaliada aqui — ela já é violação da SOLAR-ARCH-016, e 

64 # apontar o mesmo trecho duas vezes só faria ruído. 

65 if declaration.style != RouteDeclaration.REGISTRATION: 

66 return 

67 

68 if self._points_to_controller(declaration): 

69 return 

70 

71 raise SolarException( 

72 self._build_message(declaration) 

73 ) 

74 

75 def _points_to_controller( 

76 self, 

77 declaration: RouteDeclaration 

78 ) -> bool: 

79 

80 owner = self._controller_owner(declaration.target) 

81 

82 if owner is None: 

83 return False 

84 

85 return owner.endswith(self.CONTROLLER_SUFFIX) 

86 

87 def _controller_owner(self, target: str | None) -> str | None: 

88 """ 

89 Extrai o "dono" de um alvo de rota. 

90 

91 "AuthController.login" → "AuthController" 

92 "AuthController().login" → "AuthController" 

93 

94 Retorna None quando o alvo não é uma referência 

95 qualificada (ex.: lambda ou expressão inline). 

96 """ 

97 

98 # `not target` cobre `None` e `""`: lambda e expressão inline 

99 # chegam sem alvo — não são referência qualificada. 

100 if not target: 

101 return None 

102 

103 # `replace("()", "")` normaliza `AuthController().login` 

104 # para `AuthController.login`, para os dois casos terem a 

105 # mesma leitura. 

106 parts = target.replace("()", "").split(".") 

107 

108 # Menos de dois segmentos = nome solto (`login`), sem dono. 

109 if len(parts) < 2: 

110 return None 

111 

112 # `[-2]` e não `[-1]`: o último segmento é o MÉTODO 

113 # (`login`); o dono é o penúltimo (`AuthController`). 

114 return parts[-2] 

115 

116 def _build_message( 

117 self, 

118 declaration: RouteDeclaration 

119 ) -> str: 

120 

121 target = declaration.target or "<sem alvo>" 

122 

123 return ( 

124 f"{self.code()}: " 

125 f"a rota registrada em " 

126 f"{declaration.file}:{declaration.line} " 

127 f"('{declaration.mechanism}') aponta para '{target}', " 

128 f"que não é um Controller. " 

129 f"Toda rota deve apontar para um Controller " 

130 f"(ex.: UserController.create)." 

131 )

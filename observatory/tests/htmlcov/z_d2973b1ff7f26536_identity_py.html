1# Identidade da aplicação: SECRET_KEY e SOLAR_ID. 

2# 

3# É o primeiro mecanismo que o Kernel monta, e é o único ponto da 

4# arquitetura que ESCREVE no `.env`. A razão: o segredo precisa 

5# sobreviver ao reinício, e gerar uma chave nova a cada subida 

6# invalidaria toda sessão assinada com a anterior. 

7# 

8# Os dois valores têm papéis diferentes: `SECRET_KEY` é SEGREDO (assina), 

9# `SOLAR_ID` é IDENTIFICADOR (só nomeia a aplicação em log e auditoria). 

10import os 

11import secrets 

12import uuid 

13 

14 

15class Identity: 

16 """ 

17 Gerencia a identidade criptográfica e a identidade única da aplicação 

18 dentro da arquitetura SOLAR. 

19 

20 A Identity é responsável por garantir que a aplicação possua: 

21 

22 - SECRET_KEY: segredo criptográfico utilizado pela aplicação; 

23 - SOLAR_ID: identificador único e não secreto da aplicação. 

24 

25 Esses valores são persistidos no arquivo de ambiente para que sejam 

26 mantidos entre as reinicializações da aplicação. 

27 

28 A SECRET_KEY deve permanecer secreta. 

29 O SOLAR_ID não possui caráter secreto e serve para identificar 

30 unicamente a aplicação em contextos como logs, auditoria, eventos 

31 e comunicação entre serviços. 

32 """ 

33 

34 # `SECRET_KEY` e `SOLAR_ID` são os NOMES das chaves no ambiente — 

35 # não são os valores. 

36 # solar:allow-secret (nome da chave no ambiente, não o valor) 

37 SECRET_KEY = "SECRET_KEY" 

38 SOLAR_ID = "SOLAR_ID" 

39 

40 def __init__(self, env_file=".env"): 

41 # Caminho do arquivo de ambiente: injetável para o teste usar um 

42 # arquivo temporário em vez do `.env` do projeto. 

43 self.env_file = env_file 

44 

45 # `None` = "ainda não lido". É esse estado que `init` procura para 

46 # decidir se gera um valor novo. 

47 self.secret_key = None 

48 self.solar_id = None 

49 

50 def init(self): 

51 """ 

52 Inicializa a identidade da aplicação. 

53 

54 Caso SECRET_KEY ou SOLAR_ID já existam no ambiente, seus valores 

55 são preservados. 

56 

57 Caso algum deles não exista, um novo valor é gerado e persistido 

58 no arquivo de ambiente. 

59 """ 

60 

61 # Lê o que já existe: PRESERVAR o valor anterior é o ponto. Uma 

62 # chave nova a cada subida invalidaria toda sessão assinada com 

63 # a anterior. 

64 self._load_environment() 

65 

66 # `if not ...`: só gera quando não havia valor. 

67 if not self.secret_key: 

68 self.secret_key = self._generate_secret_key() 

69 

70 if not self.solar_id: 

71 self.solar_id = self._generate_solar_id() 

72 

73 # Grava SEMPRE, e não só quando gerou: se o `.env` perdeu a 

74 # linha por edição manual, ela volta. 

75 self._persist() 

76 

77 return { 

78 self.SECRET_KEY: self.secret_key, 

79 self.SOLAR_ID: self.solar_id 

80 } 

81 

82 def _load_environment(self): 

83 """ 

84 Carrega SECRET_KEY e SOLAR_ID existentes no ambiente. 

85 """ 

86 

87 self.secret_key = os.getenv(self.SECRET_KEY) 

88 self.solar_id = os.getenv(self.SOLAR_ID) 

89 

90 @staticmethod 

91 def _generate_secret_key(): 

92 """ 

93 Gera uma chave criptograficamente segura. 

94 """ 

95 

96 # `secrets` e não `random`: `random` é previsível a partir de 

97 # algumas saídas, e esta chave ASSINA sessão. 32 bytes viram 64 

98 # caracteres hexadecimais. 

99 return secrets.token_hex(32) 

100 

101 @staticmethod 

102 def _generate_solar_id(): 

103 """ 

104 Gera um identificador único para a aplicação. 

105 

106 O identificador não é um segredo. 

107 """ 

108 

109 # `uuid4` é aleatório (não deriva da máquina), e `str` porque o 

110 # valor vai para o `.env` como texto. 

111 return str(uuid.uuid4()) 

112 

113 def _persist(self): 

114 """ 

115 Persiste SECRET_KEY e SOLAR_ID no arquivo de ambiente. 

116 """ 

117 

118 # Conferir ANTES de ler: sem isso o erro sairia do `open`, com 

119 # mensagem do sistema e sem dizer qual arquivo a aplicação 

120 # esperava encontrar. 

121 if not os.path.exists(self.env_file): 

122 raise FileNotFoundError( 

123 f"O arquivo {self.env_file} não foi encontrado." 

124 ) 

125 

126 # Lê tudo para a MEMÓRIA e reescreve no fim: o arquivo é 

127 # pequeno, e editar linha a linha no disco seria mais código 

128 # para o mesmo resultado. 

129 with open(self.env_file, "r", encoding="utf-8") as file: 

130 lines = file.readlines() 

131 

132 values = { 

133 self.SECRET_KEY: self.secret_key, 

134 self.SOLAR_ID: self.solar_id 

135 } 

136 

137 # Acompanha quais chaves foram ENCONTRADAS no arquivo: o que 

138 # sobrar é acrescentado no fim. 

139 existing = set() 

140 

141 # `startswith(f"{key}=")` com o `=` JUNTO: sem ele, uma chave 

142 # `SECRET_KEY_BACKUP` casaria com `SECRET_KEY`. 

143 for index, line in enumerate(lines): 

144 

145 stripped = line.strip() 

146 

147 for key, value in values.items(): 

148 

149 if stripped.startswith(f"{key}="): 

150 # Substitui a LINHA, preservando a posição dela no 

151 # arquivo — comentários e ordem continuam legíveis. 

152 lines[index] = f"{key}={value}\n" 

153 existing.add(key) 

154 

155 for key, value in values.items(): 

156 

157 if key not in existing: 

158 lines.append(f"{key}={value}\n") 

159 

160 # `writelines` (e não `write`) porque cada item da lista já veio 

161 # com o seu `\n`. 

162 with open(self.env_file, "w", encoding="utf-8") as file: 

163 file.writelines(lines)

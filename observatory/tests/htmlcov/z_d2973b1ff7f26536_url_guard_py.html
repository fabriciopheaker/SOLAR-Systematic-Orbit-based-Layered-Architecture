1# Guarda de URL: barra destino não confiável ANTES da chamada externa. 

2# 

3# É o instrumento que a SOLAR-ARCH-030 aponta quando a URL precisa mesmo 

4# variar em tempo de execução — a regra proíbe montar URL com dado de 

5# entrada, e este é o caminho permitido para o caso legítimo. 

6# 

7# Estático e SEM REDE: não há resolução de DNS aqui (ver o docstring da 

8# classe) — resolver seria uma consulta de rede dentro do validador, e o 

9# validador precisa ser previsível. 

10from ipaddress import ip_address 

11from urllib.parse import urlsplit 

12 

13from nucleus.exceptions.security_exception import SecurityException 

14 

15 

16class UrlGuard: 

17 """ 

18 Valida uma URL ANTES de a aplicação buscar um serviço externo 

19 (SOLAR-ARCH-030). 

20 

21 É o instrumento para os casos que precisam variar em tempo de 

22 execução — em vez de a URL ir crua para o cliente HTTP: 

23 

24 UrlGuard(allowed_hosts=["api.exemplo.com"]).validate(url) 

25 requests.get(url, ...) 

26 

27 O que é barrado: 

28 

29 esquema fora da lista file:///etc/passwd 

30 endereço INTERNO http://169.254.169.254 

31 http://127.0.0.1 

32 http://10.0.0.5 

33 http://[::1] 

34 host fora da lista permitida http://outro.com 

35 credenciais na URL http://user:senha@host 

36 

37 O bloqueio de endereço interno vale para IP LITERAL, que é como 

38 a maioria dos ataques tenta alcançar o metadata service 

39 (169.254.169.254). Um NOME que resolve para endereço interno 

40 só é barrado pela lista de hosts permitidos: resolver DNS aqui 

41 seria uma consulta de rede dentro do validador. 

42 

43 Por isso a lista de hosts permitidos é a defesa principal. Sem 

44 ela, o guard ainda barra esquema, credenciais e IP interno, 

45 mas não sabe para onde um nome aponta. 

46 """ 

47 

48 # Só `https` por padrão. `http` fica de fora DE PROPÓSITO: o caso 

49 # comum é serviço externo, e `http` puro expõe a chamada. Quem 

50 # precisa de outro esquema passa `allowed_schemes`. 

51 DEFAULT_SCHEMES = ("https",) 

52 

53 # Um código por MOTIVO da recusa: quem trata a exceção distingue 

54 # "host não permitido" de "esquema errado" sem ler a mensagem. 

55 INVALID_URL = "INVALID_URL" 

56 DISALLOWED_SCHEME = "DISALLOWED_SCHEME" 

57 DISALLOWED_HOST = "DISALLOWED_HOST" 

58 PRIVATE_ADDRESS = "PRIVATE_ADDRESS" 

59 

60 # Código de erro, não credencial — o marcador diz isso à 

61 # SOLAR-ARCH-020, que acusaria o nome pelo vocabulário. 

62 # solar:allow-secret (código de erro, não credencial) 

63 URL_CREDENTIALS = "URL_CREDENTIALS" 

64 

65 def __init__( 

66 self, 

67 allowed_hosts=None, 

68 allowed_schemes=None 

69 ): 

70 # `None` significa "não há lista" — e é DIFERENTE de lista vazia, 

71 # que barraria todo host. `_validate_host` usa essa diferença 

72 # para saber se a checagem se aplica. 

73 # 

74 # `strip().lower()` na entrada: a lista é escrita à mão na 

75 # configuração, e `"API.exemplo.com "` tem que valer o mesmo 

76 # que `"api.exemplo.com"`. 

77 self.allowed_hosts = ( 

78 frozenset( 

79 host.strip().lower() 

80 for host in allowed_hosts 

81 ) 

82 if allowed_hosts is not None 

83 else None 

84 ) 

85 

86 self.allowed_schemes = frozenset( 

87 scheme.strip().lower() 

88 for scheme in (allowed_schemes or self.DEFAULT_SCHEMES) 

89 ) 

90 

91 def is_allowed(self, url) -> bool: 

92 """ 

93 Indica se a URL pode ser usada. 

94 """ 

95 

96 # Pergunta de SIM ou NÃO para quem só quer decidir sem tratar 

97 # exceção — `validate` continua sendo o caminho quando se quer o 

98 # MOTIVO. 

99 try: 

100 self.validate(url) 

101 except SecurityException: 

102 return False 

103 

104 return True 

105 

106 def validate(self, url) -> None: 

107 """ 

108 Valida a URL ou levanta SecurityException. 

109 """ 

110 

111 # Tipo antes de conteúdo: `urlsplit` de algo que não é texto 

112 # estouraria com erro de tipo, e não de segurança. 

113 if not isinstance(url, str) or not url.strip(): 

114 

115 raise SecurityException( 

116 "URL vazia ou de tipo inválido.", 

117 code=self.INVALID_URL 

118 ) 

119 

120 # `urlsplit` UMA vez, e as partes passadas adiante: cada 

121 # checagem olha um pedaço diferente da mesma análise. 

122 parts = urlsplit(url) 

123 

124 self._validate_scheme(parts) 

125 self._validate_credentials(parts) 

126 

127 # O host é extraído no meio porque as duas checagens seguintes 

128 # precisam dele — e é ele que levanta se estiver ausente. 

129 host = self._host(parts) 

130 

131 # Duas defesas em ordem: primeiro o endereço LITERAL (internos 

132 # barrados já na saída), depois a lista de permitidos. 

133 self._validate_address(host) 

134 self._validate_host(host) 

135 

136 def _validate_scheme(self, parts) -> None: 

137 

138 # `or ""`: URL sem esquema (`//host/x`) tem `scheme` vazio, e o 

139 # `None` do urlsplit estouraria na comparação. A mensagem trata 

140 # os dois casos com `<ausente>`. 

141 scheme = (parts.scheme or "").lower() 

142 

143 if scheme not in self.allowed_schemes: 

144 

145 raise SecurityException( 

146 f"Esquema '{scheme or '<ausente>'}' não permitido; " 

147 f"esperado: {sorted(self.allowed_schemes)}.", 

148 code=self.DISALLOWED_SCHEME 

149 ) 

150 

151 def _validate_credentials(self, parts) -> None: 

152 

153 # Credencial embutida na URL (`http://user:senha@host`) é 

154 # vazamento garantido: ela aparece em log, em histórico e em 

155 # mensagem de erro. 

156 if parts.username or parts.password: 

157 

158 raise SecurityException( 

159 "A URL não pode carregar credenciais.", 

160 code=self.URL_CREDENTIALS 

161 ) 

162 

163 def _host(self, parts) -> str: 

164 

165 host = (parts.hostname or "").lower() 

166 

167 if not host: 

168 

169 raise SecurityException( 

170 "A URL não informa o host.", 

171 code=self.INVALID_URL 

172 ) 

173 

174 return host 

175 

176 def _validate_address(self, host: str) -> None: 

177 """ 

178 Barra IP LITERAL que não seja público. 

179 """ 

180 

181 # `ip_address` só aceita IP LITERAL. `ValueError` é a resposta 

182 # comum — a maioria dos hosts é NOME, e nome não é resolvido 

183 # nesta checagem (ver o docstring da classe). 

184 try: 

185 address = ip_address(host) 

186 except ValueError: 

187 return 

188 

189 # `is_global` é falso para loopback, rede privada, 

190 # link-local (169.254.0.0/16), reservado e multicast. 

191 if not address.is_global: 

192 

193 raise SecurityException( 

194 f"O endereço '{host}' é interno e não pode ser " 

195 f"alcançado por requisição externa.", 

196 code=self.PRIVATE_ADDRESS 

197 ) 

198 

199 def _validate_host(self, host: str) -> None: 

200 

201 # Sem lista configurada, esta checagem não se aplica — e é isso 

202 # que o `None` distingue de lista vazia. 

203 if self.allowed_hosts is None: 

204 return 

205 

206 if self._matches_host(host): 

207 return 

208 

209 raise SecurityException( 

210 f"O host '{host}' não está na lista de permitidos.", 

211 code=self.DISALLOWED_HOST 

212 ) 

213 

214 def _matches_host(self, host: str) -> bool: 

215 """ 

216 Casa o host exato ou um subdomínio dele. 

217 """ 

218 

219 # `==` OU sufixo com PONTO: `api.exemplo.com` casa com 

220 # `exemplo.com` (subdomínio), mas `naoexemplo.com` NÃO casa — é 

221 # o ponto no `f".{allowed}"` que separa os dois casos. 

222 return any( 

223 host == allowed or host.endswith(f".{allowed}") 

224 for allowed in self.allowed_hosts 

225 )

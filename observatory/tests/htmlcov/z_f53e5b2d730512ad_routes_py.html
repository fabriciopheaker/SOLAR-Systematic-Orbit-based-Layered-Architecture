1""" 

2O ponto central das rotas da aplicação. 

3 

4A SOLAR-ARCH-018 diz onde rota pode morar: em 

5`planets/<planeta>/routes.py` ou aqui. Não existe um terceiro 

6lugar — é o que impede rota declarada no meio de um service, de um 

7repository ou de um controller. 

8 

9Este módulo faz uma coisa, e só uma: agregar os Blueprints dos 

10Planets. NÃO há rota própria aqui — até a verificação de saúde 

11pertence a um Planet (o `solar`), e é o que mantém a regra sem 

12nenhuma exceção. 

13""" 

14 

15from flask import Blueprint 

16 

17from orbits.planets.auth import routes as auth_routes 

18from orbits.planets.example import routes as example_routes 

19from orbits.planets.solar import routes as solar_routes 

20from orbits.planets.user import routes as user_routes 

21 

22 

23def routes() -> Blueprint: 

24 """ 

25 Devolve o Blueprint com as rotas da aplicação. 

26 

27 Este módulo MONTA; quem registra na aplicação é o Kernel. A 

28 separação é o que permite montar tudo em teste sem subir 

29 servidor. 

30 """ 

31 

32 # Um `Blueprint` do Flask, sem adaptador nenhum: a proteção de 

33 # entrada é responsabilidade de cada Planet, na Atmosphere dele. 

34 blueprint = Blueprint("orbits", __name__) 

35 

36 # ROTA DE SAÚDE 

37 # 

38 # Deixou de morar aqui: ela pertence ao Planet `solar`, que é o 

39 # Planet da própria plataforma. Aqui não sobra nenhuma rota 

40 # avulsa — e é isso que faz a SOLAR-ARCH-018 valer sem exceção. 

41 

42 # PLANETS 

43 # 

44 # Um Planet novo entra por uma linha DAQUI. Este é o único 

45 # lugar do projeto que conhece a lista de Planets: o Kernel 

46 # registra o conjunto, e não cada um. 

47 blueprint.register_blueprint(auth_routes.routes()) 

48 blueprint.register_blueprint(user_routes.routes()) 

49 

50 # O Planet `solar` é o da própria plataforma: responde sobre a 

51 # aplicação, e não sobre um módulo de negócio. É onde mora a 

52 # verificação de saúde. 

53 blueprint.register_blueprint(solar_routes.routes()) 

54 

55 # O Planet `example` é o fluxo completo de referência: criação de 

56 # usuário passando por todas as camadas. Ele entra aqui como 

57 # qualquer outro — é o que prova que o ponto central não precisa 

58 # conhecer o que o Planet faz. 

59 blueprint.register_blueprint(example_routes.routes()) 

60 

61 return blueprint

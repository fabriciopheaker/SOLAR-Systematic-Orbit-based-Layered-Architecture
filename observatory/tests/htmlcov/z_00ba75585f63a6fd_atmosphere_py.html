1# Mecanismo da FRONTEIRA do Planet. 

2# 

3# Quem DECLARA os guardas de entrada é a ROTA (`sanitizer=` e 

4# `validator=`, no `add_url_rule`); quem os APLICA é este mecanismo, 

5# ligado ao Blueprint do Planet por `before_request` — ver 

6# `SolarBlueprint`, que monta o mapa que este módulo consome. 

7# 

8# É um `before_request` de verdade, e não um pipeline inventado: os 

9# guardas rodam ANTES de qualquer Controller daquele Planet, e o 

10# Controller recebe a requisição já tratada. Nenhum Controller 

11# conhece Sanitizer, Validator ou RequestPipeline. 

12from flask import g, request 

13 

14from nucleus.contracts.http_input import HttpInput 

15from nucleus.middleware.request_pipeline import RequestPipeline 

16 

17 

18# Onde o corpo tratado espera até o Controller pedir. 

19# 

20# Fica no `g` do Flask, que é o contexto da REQUISIÇÃO: duas 

21# requisições simultâneas não se enxergam, o que um atributo de 

22# módulo não garantiria. 

23# 

24# O VALOR é o nome que o Controller escreve — `g.solar_data`. Está 

25# aqui para haver um lugar só que decide a chave, e o Controller a 

26# cita direto, sem import nenhum entre a fronteira e ele. 

27TREATED_DATA = "solar_data" 

28 

29# Uma instância só, criada na importação. 

30# 

31# O pipeline não guarda estado entre requisições, e recriá-lo a cada 

32# chamada montaria um `RuleEngine` por requisição — trabalho que não 

33# muda nada no resultado. 

34_PIPELINE = RequestPipeline() 

35 

36 

37def apply(entries): 

38 """ 

39 Devolve a função de fronteira do Planet. 

40 

41 `entries` mapeia ENDPOINT → (Sanitizer, Validator). Quem o monta é 

42 o `SolarBlueprint`, a partir dos guardas que cada ROTA declarou: 

43 

44 # planets/example/routes.py 

45 blueprint = SolarBlueprint("example", __name__, ...) 

46 

47 blueprint.add_url_rule( 

48 "/users", 

49 "create_example_user", 

50 CreateUserController.create, 

51 methods=["POST"], 

52 sanitizer=CreateUserSanitizer, 

53 validator=CreateUserValidator, 

54 ) 

55 

56 O que o mecanismo liga é o resultado: 

57 

58 self.before_request(apply(self._guards)) 

59 

60 Endpoint que NÃO está no mapa passa direto: é rota que não 

61 transporta dados (um GET, por exemplo), e não há o que validar 

62 nem o que limpar. 

63 """ 

64 

65 def protect(): 

66 

67 # SEMPRE presente, mesmo quando a rota não tem guarda 

68 # declarado: o Controller lê `g.solar_data` sem se preocupar, 

69 # e rota fora do mapa recebe vazio em vez de um 

70 # `AttributeError` que apareceria longe da causa. 

71 setattr(g, TREATED_DATA, {}) 

72 

73 guards = entries.get(local_endpoint()) 

74 

75 if guards is None: 

76 return None 

77 

78 sanitizer, validator = guards 

79 

80 # `silent=True`: corpo ausente ou malformado vira `None`, e 

81 # quem recusa é o Validator — com código e mensagem de 

82 # contrato, em vez de a biblioteca estourar antes, com erro 

83 # dela. 

84 payload = request.get_json(silent=True) 

85 

86 # `carries_data=True` explícito: estar no mapa JÁ É declarar 

87 # que esta entrada transporta dados. Por isso o `content_type` 

88 # não aparece em nenhum lugar — JSON é o normal, e o caso 

89 # especial (upload, form) tem guarda próprio no mapa. 

90 http_input = HttpInput( 

91 method=request.method, 

92 name=request.path, 

93 sanitizer=sanitizer, 

94 validator=validator, 

95 carries_data=True, 

96 ) 

97 

98 # A ordem (validar → sanitizar) é do RequestPipeline, e é o 

99 # motivo de ele existir: um lugar só decide isso. 

100 setattr( 

101 g, 

102 TREATED_DATA, 

103 _PIPELINE.handle(http_input, payload) 

104 ) 

105 

106 return None 

107 

108 return protect

111def local_endpoint() -> str: 

112 """ 

113 O endpoint da rota SEM o prefixo do Blueprint. 

114 

115 Registrado num Blueprint, o Flask prefixa o endpoint com o nome 

116 dele — a rota de `/examples/users` vira 

117 `example.create_example_user`. O mapa da Atmosphere usa o nome 

118 LOCAL (o que está escrito no `add_url_rule`), porque é o nome que 

119 o Planet escreveu e conhece. 

120 

121 Um Blueprint aninhado (`orbits.example.create_example_user`) 

122 também funciona: o nome local é sempre o ÚLTIMO segmento. 

123 """ 

124 

125 endpoint = request.endpoint or "" 

126 

127 return endpoint.rsplit(".", 1)[-1]

1# Leitura e normalização das variáveis de ambiente. 

2# 

3# É um MECANISMO do Nucleus, e não uma camada: o que ele faz é ler o 

4# `.env` do ambiente ativo e devolver um dicionário de configuração para 

5# o Kernel montar a aplicação. Ele não conhece nenhum componente da 

6# arquitetura. 

7import os 

8 

9from dotenv import load_dotenv 

10 

11from nucleus.exceptions.solar_exception import SolarException 

12 

13 

14class Environment: 

15 """ 

16 Lê e normaliza as variáveis de ambiente da aplicação. 

17 

18 O Environment apenas lê e normaliza as variáveis do arquivo .env. 

19 

20 Ele NÃO monta a URL de conexão, NÃO decide o dialect/driver e 

21 NÃO interpreta as configurações específicas de banco de dados. 

22 

23 Essas responsabilidades pertencem ao DriverSQLAlchemy. 

24 """ 

25 

26 # Os três ambientes conhecidos. Um `APP_ENV` fora daqui é ERRO de 

27 # configuração, e não um valor a ser aceito em silêncio — daí o 

28 # `raise` em `init`. 

29 SUPPORTED_ENVIRONMENTS = {'DEV', 'TEST', 'PROD'} 

30 

31 @staticmethod 

32 def init(): 

33 

34 """ 

35 Configura a aplicação com base no ambiente, 

36 carregando e normalizando as variáveis correspondentes 

37 do arquivo .env. 

38 """ 

39 

40 # Carrega o `.env` para o processo. Sem isso, o `os.getenv` da 

41 # linha seguinte devolveria `None` e a aplicação nem saberia 

42 # dizer em que ambiente está. 

43 load_dotenv() 

44 

45 environment = os.getenv('APP_ENV') 

46 

47 if environment not in Environment.SUPPORTED_ENVIRONMENTS: 

48 raise SolarException( 

49 f"Ambiente '{environment}' não suportado.", 

50 code="INVALID_ENVIRONMENT" 

51 ) 

52 

53 # Dicionário de configuração: é ele que o Kernel guarda e que o 

54 # resto do Nucleus consulta por chave. As chaves são NOMES 

55 # LÓGICOS (`DB_HOST`), e não as variáveis do `.env` 

56 # (`DEV_DB_HOST`) — o prefixo do ambiente fica só na leitura. 

57 return { 

58 'APP_ENV': environment, 

59 

60 # Configuração bruta do banco de dados. 

61 # A interpretação pertence ao DriverSQLAlchemy. 

62 'DB_CONNECTION': Environment.__read( 

63 environment, 'DB_CONNECTION' 

64 ), 

65 

66 'DB_NAME': Environment.__read( 

67 environment, 'DB_NAME' 

68 ), 

69 

70 'DB_HOST': Environment.__read( 

71 environment, 'DB_HOST' 

72 ), 

73 

74 'DB_PORT': Environment.__read( 

75 environment, 'DB_PORT' 

76 ), 

77 

78 'DB_USER': Environment.__read( 

79 environment, 'DB_USER' 

80 ), 

81 

82 'DB_PASS': Environment.__read( 

83 environment, 'DB_PASS' 

84 ), 

85 

86 'CORS': Environment.__read( 

87 environment, 'CORS_ORIGINS' 

88 ), 

89 

90 'PORT': Environment.__read( 

91 environment, 'PORT' 

92 ), 

93 

94 'HOST': Environment.__read( 

95 environment, 'HOST' 

96 ), 

97 } 

98 

99 @staticmethod 

100 def __read(environment: str, key: str): 

101 """ 

102 Lê e normaliza uma variável do ambiente atual. 

103 

104 Valores vazios são normalizados para None. 

105 """ 

106 

107 # O PREFIXO do ambiente é o que permite os três estarem no mesmo 

108 # `.env`: `DEV_DB_HOST`, `TEST_DB_HOST`, `PROD_DB_HOST`. 

109 value = os.getenv(f"{environment}_{key}") 

110 

111 if value is None: 

112 return None 

113 

114 # `strip()`: espaço em volta de valor no `.env` é acidente de 

115 # digitação, e sem isso a URL de conexão sairia com espaço no 

116 # meio. 

117 value = value.strip() 

118 

119 # `value or None`: variável DECLARADA mas VAZIA vira `None`. É o 

120 # caso de `PROD_HOST=` — declarada para ficar visível, ainda sem 

121 # valor. 

122 return value or None

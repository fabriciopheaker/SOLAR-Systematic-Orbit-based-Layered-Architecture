145@dataclass(frozen=True) 

146class Check: 

147 """ 

148 Uma verificação: um SCANNER e as regras que falam do contexto 

149 que ele produz. 

150 

151 O agrupamento não é decorativo — cada contexto tem o seu 

152 RuleEngine, e passar um alvo de um contexto para um 

153 RuleEngine de outro quebra a execução. 

154 

155 name: 

156 Nome do contexto (ex.: "Dependency"). 

157 

158 scanner: 

159 Classe do scanner; recebe a raiz do projeto. 

160 

161 rules: 

162 Regras que entendem do contexto desse scanner. 

163 """ 

164 

165 # Nome do contexto; é ele que aparece no relatório e no 

166 # `check` da linha de comando. 

167 name: str 

168 

169 # A CLASSE do scanner, não a instância: cada check recebe a 

170 # raiz do projeto no momento de varrer, e a mesma classe serve 

171 # a varreduras diferentes. 

172 scanner: type 

173 

174 # Regras que sabem ler o contexto produzido por esse scanner. 

175 # Passar um alvo de outro contexto para estas regras quebra a 

176 # execução — o RuleEngine é por contexto. 

177 rules: tuple[Rule, ...]

265class ProjectValidator: 

266 """ 

267 Executa TODAS as regras arquiteturais sobre o projeto. 

268 

269 É o ponto de execução que faltava: os scanners e as regras 

270 existiam, mas nada os invocava fora dos testes. 

271 

272 report = ProjectValidator(Path(".")).validate() 

273 

274 if not report.is_clean(): 

275 print(report) 

276 

277 Diferença em relação ao RuleEngine: o RuleEngine PARA na 

278 primeira violação (o contrato é "validar ou levantar"). Aqui a 

279 intenção é o contrário — mostrar TUDO o que está fora do lugar 

280 de uma vez, para não transformar a correção em um jogo de 

281 descobre-e-corrige. Por isso cada regra é chamada 

282 individualmente e a exceção é recolhida como violação. 

283 

284 Um erro que NÃO seja SolarException não é recolhido: seria bug 

285 na regra ou no scanner, e deve aparecer como erro. 

286 """ 

287 

288 def __init__(self, root: Path, checks=None): 

289 # `resolve()` e não `Path(root)`: o scanner subtrai a raiz 

290 # do caminho de cada arquivo para gravar o local RELATIVO 

291 # no relatório. Sem o caminho absoluto normalizado, essa 

292 # subtração falha quando o `check` roda de outro diretório. 

293 self.root = Path(root).resolve() 

294 

295 # `checks` injetável: os testes passam um check só para 

296 # exercitar uma regra sem varrer o projeto todo. 

297 self.checks = ( 

298 project_checks() 

299 if checks is None 

300 else tuple(checks) 

301 ) 

302 

303 def validate(self) -> ProjectReport: 

304 """ 

305 Varre o projeto e devolve o relatório. 

306 """ 

307 

308 results: list[CheckResult] = [] 

309 rules = 0 

310 

311 for check in self.checks: 

312 

313 # Conta antes de rodar: a contagem é do que FOI 

314 # aplicado, e um check que levante por bug do scanner 

315 # já teria aplicado as suas regras. 

316 rules += len(check.rules) 

317 

318 # Um resultado por check, na ordem declarada em 

319 # `project_checks()`. 

320 results.append( 

321 CheckResult( 

322 name=check.name, 

323 violations=tuple(self._check(check)) 

324 ) 

325 ) 

326 

327 return ProjectReport( 

328 results=tuple(results), 

329 rules=rules 

330 )

332 def _check(self, check: Check) -> list[Violation]: 

333 """ 

334 Roda as regras de um check sobre cada alvo do seu scanner. 

335 """ 

336 

337 violations: list[Violation] = [] 

338 

339 # O scanner lê o projeto inteiro uma vez e devolve TODOS 

340 # os alvos do contexto; as regras do check rodam sobre 

341 # cada um deles. 

342 for target in check.scanner(self.root).scan(): 

343 

344 violations.extend( 

345 self._apply(check, target) 

346 ) 

347 

348 return violations 

349 

350 def _apply( 

351 self, 

352 check: Check, 

353 target 

354 ) -> list[Violation]: 

355 """ 

356 Aplica as regras do check a um alvo. 

357 """ 

358 

359 violations: list[Violation] = [] 

360 

361 for rule in check.rules: 

362 

363 # `try` por regra, e não em volta do laço: uma 

364 # violação não pode impedir as regras seguintes de 

365 # falarem. O objetivo aqui é listar TUDO o que está 

366 # fora do lugar de uma vez. 

367 try: 

368 rule.validate(target) 

369 

370 # Só `SolarException` é recolhida — ela significa 

371 # "o projeto viola a regra". Qualquer outro erro é bug 

372 # na regra ou no scanner e DEVE subir como erro: 

373 # engolir aqui esconderia defeito do próprio 

374 # validador. 

375 except SolarException as error: 

376 

377 violations.append( 

378 Violation( 

379 rule=rule.code(), 

380 description=rule.description(), 

381 message=str(error), 

382 location=_location(target) 

383 ) 

384 ) 

385 

386 return violations

389def _location(target) -> str: 

390 """ 

391 Local do alvo, quando o contexto informa. 

392 

393 Os contextos carregam o local de formas diferentes 

394 (`file`/`line`, `path`, `name`), e alguns não carregam nada — 

395 a mensagem da regra continua dizendo onde é. 

396 """ 

397 

398 # `getattr` com padrão: os contextos são dataclasses 

399 # diferentes e nem todos carregam local. Perguntar o que 

400 # existe é mais barato que obrigar todo contexto a ter os 

401 # mesmos campos. 

402 file = getattr(target, "file", None) 

403 line = getattr(target, "line", None) 

404 

405 # Linha junto do arquivo quando houver: "arquivo:linha" é o 

406 # formato que o terminal e o editor entendem como clique. 

407 if file and line is not None: 

408 return f"{file}:{line}" 

409 

410 if file: 

411 return str(file) 

412 

413 # Ordem de preferência: `path` é o local real; `name` é o 

414 # último recurso (ex.: contextos de estrutura, que só têm o 

415 # nome do Planet). 

416 for attribute in ("path", "name"): 

417 

418 value = getattr(target, attribute, None) 

419 

420 if value: 

421 return str(value) 

422 

423 return ""

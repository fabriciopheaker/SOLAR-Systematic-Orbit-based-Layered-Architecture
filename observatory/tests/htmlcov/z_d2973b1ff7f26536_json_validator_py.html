1# Validador de corpo JSON. 

2# 

3# É a guarda de ENTRADA (SOLAR-ARCH-013) e o PRIMEIRO passo do 

4# pipeline: confere que os campos obrigatórios CHEGARAM. O 

5# sanitizador roda depois, sobre dado que se sabe completo — não há 

6# o que limpar no campo que não veio. 

7# 

8# ESCOPO: estrutura, não negócio. Aqui se responde "o corpo tem os 

9# campos que a entrada exige?". "O e-mail tem formato válido?" é do 

10# UseCase, e "o usuário já existe?" é do banco. 

11 

12from collections.abc import Mapping 

13 

14from nucleus.contracts.validator import Validator 

15from nucleus.exceptions.validation_exception import ValidationException 

16from nucleus.notication.notification import Notification 

17 

18 

19class JsonValidator(Validator): 

20 """ 

21 Confere os campos OBRIGATÓRIOS de um corpo JSON. 

22 

23 JsonValidator(required=("name", "email")) 

24 

25 Um campo obrigatório está faltando em três casos, e os três são a 

26 mesma coisa para quem consome o dado: 

27 

28 - a chave não veio; 

29 - veio com `None`; 

30 - veio como texto vazio ou só de espaços. 

31 

32 Separá-los daria três mensagens para o mesmo problema. 

33 

34 Recolhe TODOS os campos que faltam antes de recusar: quem está 

35 corrigindo a chamada precisa da lista inteira, e não de uma 

36 tentativa por campo. 

37 

38 É ESPECÍFICO do tipo de conteúdo `application/json`, como manda o 

39 contrato: form data e arquivo têm o seu. 

40 """ 

41 

42 INVALID_PAYLOAD = "INVALID_JSON_BODY" 

43 MISSING_FIELDS = "MISSING_FIELDS" 

44 

45 def __init__(self, required=()): 

46 # Tupla, e não lista: a exigência não muda depois de criada. 

47 # Um validador que muda de regra no meio da requisição seria 

48 # imprevisível. 

49 self.required = tuple(required) 

50 

51 def validate(self, payload) -> None: 

52 """ 

53 Levanta `ValidationException` quando o corpo não serve. 

54 """ 

55 

56 if not isinstance(payload, Mapping): 

57 raise ValidationException( 

58 "O corpo da requisição deve ser um objeto JSON.", 

59 code=self.INVALID_PAYLOAD 

60 ) 

61 

62 # O Notification é o acumulador do projeto: ele existe 

63 # exatamente para relatar vários problemas de uma vez. 

64 notification = Notification() 

65 

66 for field in self.required: 

67 

68 if self._present(payload, field): 

69 continue 

70 

71 notification.add( 

72 code=self.MISSING_FIELDS, 

73 message=f"O campo '{field}' é obrigatório.", 

74 field=field 

75 ) 

76 

77 if not notification.has_errors(): 

78 return 

79 

80 # UMA exceção com a lista. O `code` é estável (quem trata não 

81 # lê a mensagem), e a mensagem junta os problemas. 

82 # 

83 # O que sai para o cliente NÃO é esta mensagem: o 

84 # ErrorHandler responde com mensagem genérica e guarda o 

85 # detalhe no log (SOLAR-ARCH-028). 

86 raise ValidationException( 

87 self._message(notification), 

88 code=self.MISSING_FIELDS 

89 ) 

90 

91 @staticmethod 

92 def _present(payload: Mapping, field: str) -> bool: 

93 """ 

94 Indica se o campo veio COM valor. 

95 """ 

96 

97 value = payload.get(field) 

98 

99 if value is None: 

100 return False 

101 

102 # Texto só de espaço é o mesmo que vazio — e é o que se 

103 # escreve quando o formulário manda o campo em branco. 

104 if isinstance(value, str): 

105 return bool(value.strip()) 

106 

107 return True 

108 

109 @staticmethod 

110 def _message(notification: Notification) -> str: 

111 """ 

112 Junta os problemas em uma frase. 

113 """ 

114 

115 return " ".join( 

116 error["message"] for error in notification.all() 

117 )
